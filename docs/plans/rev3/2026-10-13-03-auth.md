# 03 · Servicio `auth` — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Autenticación del administrador con usuario + contraseña (Argon2id) + TOTP, con bloqueo y retraso progresivo, sesiones revocables y el endpoint `verify-admin` que usa el gateway (riesgos R4 y R5). El secreto TOTP queda cifrado con una KEK rotable (R2).

**Architecture:**
- Rutas públicas: `/api/auth/login`, `/totp`, `/logout`, `/me`, con `require_gateway`.
- Una ruta interna para el `auth_request` de Nginx: `/internal/verify-admin`, con `allow("gateway")`.
- Sesiones en la tabla `admin_sessions`: etapa `preauth` y luego `full`, con un token nuevo al completar el TOTP.
- Eventos a `audit` vía `AuditClient`; `admin.login.ok` es obligatorio.
- Un CLI crea admins y otro rota la KEK.

**Tech Stack:** Python 3.13, FastAPI, SQLAlchemy async, Alembic, argon2-cffi (vía `shared.passwords`), pyotp, qrcode, pytest.

**Spec:** `docs/specs/2026-10-12-chat-seguro-design.md` §7.1, §8.1–§8.2, §9.1, §11, §14.4, §16.4 (R2, R4, R5, R11, R16).

## Global Constraints

Las del índice (`2026-10-13-00-indice.md`). En particular:
- `text()` con parámetros nombrados.
- Todas las rutas exigen `X-Service-Token` salvo `/health`.
- Errores `{"error": "..."}` genéricos.
- Cookies `__Host-admin` con `Secure`, `HttpOnly` y `SameSite=Strict`.
- Argon2id con ≤ 4 cálculos concurrentes.
- Commits en español, sin atribución.

## Review Focus

Ningún caso del índice cae aquí. A vigilar: el usuario se normaliza a minúsculas tanto al crear el admin como al hacer login (`ROOT` y `root` son el mismo); lo cubre `test_usuario_en_mayusculas_es_el_mismo`.

import time

import httpx
from fastapi import FastAPI
from sqlalchemy.ext.asyncio import AsyncEngine, create_async_engine

from auth_svc import routes
from auth_svc.config import Settings, load_settings
from shared.audit_client import AuditClient
from shared.service_auth import ServiceAuth, ServiceClient
from shared.web import create_base_app


def create_app(
    settings: Settings | None = None,
    *,
    engine: AsyncEngine | None = None,
    service_auth: ServiceAuth | None = None,
    transport: httpx.AsyncBaseTransport | None = None,
) -> FastAPI:
    settings = settings or load_settings()
    service_auth = service_auth or ServiceAuth.from_secrets("auth", needs_token=True)
    peers = ServiceClient(service_auth, {"audit": settings.audit_url}, transport)
    app = create_base_app(
        title="auth",
        settings=settings,
        engine=engine or create_async_engine(settings.database_url, pool_pre_ping=True),
        service_auth=service_auth,
        audit=AuditClient(peers),
    )
    app.state.peers = peers
    app.state.clock = time.time
    app.include_router(routes.router)
    return app

"""Alta de administradores (spec §9.1, paso 7). No existen credenciales por defecto."""

import re
from uuid import uuid4

import pyotp
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncEngine

from auth_svc.config import Settings
from shared.crypto import seal, totp_aad
from shared.passwords import hash_secret_async, strong_enough

USERNAME_RE = re.compile(r"[a-z0-9_.-]{3,32}")


async def create_admin(engine: AsyncEngine, settings: Settings, audit, username: str, password: str) -> str:
    """Crea el admin y devuelve su secreto TOTP (base32) para mostrarlo una sola vez."""
    username = username.strip().lower()
    if not USERNAME_RE.fullmatch(username):
        raise ValueError("usuario inválido: 3–32 caracteres a-z, 0-9, '_', '.' o '-'")
    if not strong_enough(password, forbid=username):
        raise ValueError(
            "contraseña débil: ≥ 12 caracteres y 3 tipos (minúsculas, mayúsculas, dígitos, símbolos)"
        )
    admin_id, secret = str(uuid4()), pyotp.random_base32()
    password_hash = await hash_secret_async(password)
    async with engine.begin() as conn:
        await conn.execute(
            text(
                "INSERT INTO admins (id, username, password_hash, totp_secret, kek_version) "
                "VALUES (:id, :username, :hash, :totp, :version)"
            ),
            {
                "id": admin_id,
                "username": username,
                "hash": password_hash,
                "totp": seal(settings.kek, secret.encode(), totp_aad(admin_id)),
                "version": settings.current_kek_version,
            },
        )
        await audit.emit("info", "admin.create", actor="system", target=username, required=True)
    return secret
```

"""Crea un administrador e imprime su QR de TOTP (spec §9.1, paso 7).

Uso dentro del contenedor:  python -m auth_svc.create_admin <usuario> [--password-stdin]
"""

import argparse
import asyncio
import sys
from getpass import getpass

import pyotp
import qrcode
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import create_async_engine

from auth_svc.admins import create_admin
from auth_svc.config import load_settings
from shared.audit_client import AuditClient
from shared.service_auth import ServiceAuth, ServiceClient


def read_password(from_stdin: bool) -> str:
    if from_stdin:
        return sys.stdin.readline().rstrip("\n")
    password = getpass("Contraseña: ")
    if password != getpass("Repite la contraseña: "):
        sys.exit("las contraseñas no coinciden")
    return password


async def run(username: str, password: str) -> str:
    settings = load_settings()
    engine = create_async_engine(settings.database_url)
    peers = ServiceClient(ServiceAuth.from_secrets("auth", needs_token=True), {"audit": settings.audit_url})
    try:
        return await create_admin(engine, settings, AuditClient(peers), username, password)
    finally:
        await engine.dispose()


def main() -> None:
    parser = argparse.ArgumentParser(description="Crea un administrador con TOTP")
    parser.add_argument("username")
    parser.add_argument("--password-stdin", action="store_true", help="lee la contraseña de stdin")
    args = parser.parse_args()
    try:
        secret = asyncio.run(run(args.username, read_password(args.password_stdin)))
    except ValueError as exc:
        sys.exit(str(exc))
    except IntegrityError:
        sys.exit("ese usuario ya existe")
    uri = pyotp.TOTP(secret).provisioning_uri(name=args.username.strip().lower(), issuer_name="Chat Seguro")
    qr = qrcode.QRCode(border=1)
    qr.add_data(uri)
    qr.print_ascii(invert=True)
    print("Escanea el QR con tu app TOTP o ingresa el secreto manualmente. No se volverá a mostrar.")
    print(f"TOTP_SECRET={secret}")


if __name__ == "__main__":
    main()
```

"""Rutas de auth (spec §8, §9.1; riesgos 4 y 5)."""

import asyncio

from fastapi import APIRouter, Depends, HTTPException, Request, Response
from pydantic import BaseModel, Field
from sqlalchemy import text

from shared import ratelimit
from shared.crypto import new_token, token_hash
from shared.identity import client_ip, require_gateway
from shared.passwords import verify_secret_async
from shared.web import ADMIN_COOKIE, set_cookie

router = APIRouter()
PREAUTH_SECONDS = 300


class Credentials(BaseModel):
    username: str = Field(min_length=1, max_length=64)
    password: str = Field(min_length=1, max_length=256)


def login_limits(settings, username: str, ip: str | None) -> list[tuple[str, int]]:
    return [(f"user:{username}", settings.login_max_fails), (f"ip:{ip}", settings.login_max_fails)]


def too_many(settings) -> HTTPException:
    return HTTPException(
        429, "demasiados intentos", headers={"Retry-After": str(settings.login_window_min * 60)}
    )


@router.post("/api/auth/login", status_code=204, dependencies=[Depends(require_gateway)])
async def login(body: Credentials, request: Request, response: Response) -> None:
    settings, engine, audit = request.app.state.settings, request.app.state.engine, request.app.state.audit
    ip, username = client_ip(request), body.username.strip().lower()
    limits = login_limits(settings, username, ip)
    async with engine.connect() as conn:
        locked = await ratelimit.any_over(conn, limits, settings.login_window_min)
        admin = (

"""TOTP (RFC 6238) con anti-reuso: un código vale una sola vez (spec §9.1, paso 5)."""

import hmac

import pyotp

STEP_SECONDS = 30


def verify_totp(secret: str, code: str, last_step: int, now: float) -> int | None:
    """Devuelve el paso de tiempo aceptado (ventana ±1) o None. Rechaza pasos ya usados."""
    totp = pyotp.TOTP(secret)
    current = int(now) // STEP_SECONDS
    for step in (current - 1, current, current + 1):
        if step > last_step and hmac.compare_digest(totp.at(step * STEP_SECONDS), code):
            return step
    return None
```

"""Re-envuelve los secretos TOTP con la KEK vigente (spec §11.3).

Procedimiento: montar kek_auth_v{n+1}, poner CURRENT_KEK_VERSION=n+1, reiniciar auth y ejecutar
    docker compose -f deploy/docker-compose.yml exec auth python -m auth_svc.rotate_kek
Después ya se puede retirar el secreto de la versión anterior.
"""

import asyncio

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncEngine, create_async_engine

from auth_svc.config import Settings, load_settings
from shared.audit_client import AuditClient
from shared.crypto import seal, totp_aad, unseal
from shared.service_auth import ServiceAuth, ServiceClient


async def rotate(engine: AsyncEngine, settings: Settings, audit) -> int:
    target = settings.current_kek_version
    async with engine.begin() as conn:
        rows = (
            await conn.execute(
                text("SELECT id, totp_secret, kek_version FROM admins WHERE kek_version <> :v FOR UPDATE"),
                {"v": target},
            )
        ).all()
        for row in rows:
            aad = totp_aad(str(row.id))
            secret = unseal(settings.keks[row.kek_version], bytes(row.totp_secret), aad)
            await conn.execute(
                text("UPDATE admins SET totp_secret = :secret, kek_version = :v WHERE id = :id"),
                {"secret": seal(settings.kek, secret, aad), "v": target, "id": row.id},
            )
        if rows:
            await audit.emit(
                "info", "key.rotate.auth", actor="system", details={"to": target, "admins": len(rows)},
                required=True,
            )
    return len(rows)


async def _main() -> int:
    settings = load_settings()
    engine = create_async_engine(settings.database_url)
    peers = ServiceClient(ServiceAuth.from_secrets("auth", needs_token=True), {"audit": settings.audit_url})
    try:
        return await rotate(engine, settings, AuditClient(peers))
    finally:
        await engine.dispose()


if __name__ == "__main__":
    count = asyncio.run(_main())
    print(f"{count} secretos TOTP re-envueltos con la KEK vigente")
```
