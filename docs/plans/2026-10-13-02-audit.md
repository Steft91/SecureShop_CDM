# 02 · Servicio `audit` — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Servicio que recibe los eventos de seguridad de los demás servicios, los encadena con SHA-256 en su propia BD (rol con solo `INSERT, SELECT`) y permite al admin consultarlos y verificar la integridad de la cadena (riesgo R3).

**Architecture:** FastAPI sobre `shared.web.create_base_app`.
- `POST /internal/events` es la única escritura: acepta solo a auth, rooms, chat y media, y el campo `service` se deduce del token, nunca del payload.
- `GET /api/admin/audit` y `/verify` las sirve el gateway con identidad de admin.
- audit registra sus propios rechazos escribiendo directo en su BD (`LocalAudit`).

**Tech Stack:** Python 3.13, FastAPI, SQLAlchemy async + asyncpg, Alembic, PostgreSQL 17, pytest.

**Spec:** `docs/specs/2026-10-12-chat-seguro-design.md` §7.5, §7.6, §8.2, §14, §16.4 (R3, R11, R16), §21 (RNF-07, RNF-19).

> **Revisión 4 (2026-10-20, spec rev. 4).** AUD-1…AUD-3 no cambian. **AUD-4, nueva:** el `request_id` de cada evento queda dentro de la cadena de hash, y una prueba mide la consulta por severidad con 100 000 eventos (≤ 2 s).

## Global Constraints

Las del índice (`2026-10-13-00-indice.md`). En particular:
- `text()` con parámetros nombrados.
- Todas las rutas exigen `X-Service-Token` salvo `/health`.
- Errores `{"error": "..."}`.
- Un test `test_rNN_*` por riesgo.
- Commits en español, sin atribución.

## Review Focus

Ningún caso del índice cae aquí. Riesgo propio a vigilar: un `details` con `NaN` o enorme no debe romper la cadena. Lo cubre `test_detalles_con_nan_se_rechazan` en AUD-2.

`services/audit/audit_svc/chain.py`:
```python
"""Cadena de hash del log de auditoría (spec §14.1, riesgo 3)."""

import hashlib
import ipaddress
import json
from datetime import UTC, datetime

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncConnection

GENESIS = bytes(32)
# ponytail: un candado global serializa las escrituras para encadenar sin carreras;
# a esta escala no importa. Si el volumen crece, encadenar por lotes.
LOCK_KEY = 42


def normalize_ip(ip: str | None) -> str | None:
    if not ip:
        return None
    try:
        return str(ipaddress.ip_address(ip))
    except ValueError:
        return None


def digest(prev: bytes, row: dict) -> bytes:
    canonical = json.dumps(row, sort_keys=True, separators=(",", ":"), ensure_ascii=False)
    return hashlib.sha256(prev + canonical.encode()).digest()


async def append(
    conn: AsyncConnection,
    *,
    service: str,
    severity: str,
    action: str,
    actor: str,
    target: str | None = None,
    ip: str | None = None,
    details: dict | None = None,
) -> int:
    await conn.execute(text("SELECT pg_advisory_xact_lock(:key)"), {"key": LOCK_KEY})
    last = (await conn.execute(text("SELECT hash FROM audit_log ORDER BY id DESC LIMIT 1"))).scalar()
    prev = bytes(last) if last is not None else GENESIS
    ts = datetime.now(UTC)
    row = {
        "ts": ts.isoformat(),
        "service": service,
        "severity": severity,
        "actor": actor,
        "action": action,
        "target": target,
        "ip": normalize_ip(ip),
        "details": details or {},
    }
    result = await conn.execute(
        text(
            "INSERT INTO audit_log "
            "(ts, service, severity, actor, action, target, ip, details, prev_hash, hash) "
            "VALUES (:ts, :service, :severity, :actor, :action, :target, "
            "CAST(CAST(:ip AS text) AS inet), CAST(CAST(:details AS text) AS jsonb), :prev, :hash) "
            "RETURNING id"
        ),
        {
            **row,
            "ts": ts,
            "details": json.dumps(row["details"], ensure_ascii=False),
            "prev": prev,
            "hash": digest(prev, row),
        },
    )
    return result.scalar_one()


async def verify(conn: AsyncConnection) -> int | None:
    """Recorre la cadena; devuelve el id de la primera fila alterada o None si está íntegra."""
    result = await conn.execute(
        text(
            "SELECT id, ts, service, severity, actor, action, target, host(ip) AS ip, "
            "details::text AS details, prev_hash, hash FROM audit_log ORDER BY id"
        )
    )
    prev = GENESIS
    for record in result.mappings():
        row = {
            "ts": record["ts"].isoformat(),
            "service": record["service"],
            "severity": record["severity"],
            "actor": record["actor"],
            "action": record["action"],
            "target": record["target"],
            "ip": record["ip"],
            "details": json.loads(record["details"]),
        }
        if bytes(record["prev_hash"]) != prev or bytes(record["hash"]) != digest(prev, row):
            return record["id"]
        prev = bytes(record["hash"])
    return None
```

"""audit registra sus propios eventos directo en su BD: no se llama a sí mismo por HTTP."""

from sqlalchemy.ext.asyncio import AsyncEngine

from audit_svc import chain


class LocalAudit:
    def __init__(self, engine: AsyncEngine) -> None:
        self.engine = engine

    async def emit(
        self,
        severity: str,
        action: str,
        *,
        actor: str,
        target: str | None = None,
        ip: str | None = None,
        details: dict | None = None,
        required: bool = False,
    ) -> None:
        async with self.engine.begin() as conn:
            await chain.append(
                conn, service="audit", severity=severity, action=action, actor=actor,
                target=target, ip=ip, details=details,
            )
```

"""Rutas de audit (spec §8.1–§8.2, §14)."""

import json
from typing import Literal

from fastapi import APIRouter, Depends, Request
from pydantic import BaseModel, Field, field_validator

from audit_svc import chain
from shared.service_auth import allow

router = APIRouter()


class EventIn(BaseModel):
    severity: Literal["info", "warning", "critical"]
    action: str = Field(pattern=r"^[a-z0-9_.]{3,64}$")
    actor: str = Field(min_length=1, max_length=128)
    target: str | None = Field(default=None, max_length=512)
    ip: str | None = Field(default=None, max_length=64)
    details: dict = Field(default_factory=dict)

    @field_validator("details")
    @classmethod
    def details_serializables(cls, value: dict) -> dict:
        encoded = json.dumps(value, allow_nan=False)  # NaN/Infinity no caben en jsonb
        if len(encoded) > 8192:
            raise ValueError("details demasiado grande")
        return value


@router.post(
    "/internal/events",
    status_code=201,
    dependencies=[Depends(allow("auth", "rooms", "chat", "media"))],
)
async def ingest(event: EventIn, request: Request) -> dict:
    async with request.app.state.engine.begin() as conn:
        event_id = await chain.append(conn, service=request.state.caller, **event.model_dump())
    return {"id": event_id}
```

"""audit registra sus propios eventos directo en su BD: no se llama a sí mismo por HTTP."""

from sqlalchemy.ext.asyncio import AsyncEngine

from audit_svc import chain
from shared.observability import REQUEST_ID


class LocalAudit:
    def __init__(self, engine: AsyncEngine) -> None:
        self.engine = engine

    async def emit(
        self,
        severity: str,
        action: str,
        *,
        actor: str,
        target: str | None = None,
        ip: str | None = None,
        details: dict | None = None,
        required: bool = False,
    ) -> None:
        details = dict(details or {})
        if REQUEST_ID.get():
            details["request_id"] = REQUEST_ID.get()
        async with self.engine.begin() as conn:
            await chain.append(
                conn,
                service="audit",
                severity=severity,
                action=action,
                actor=actor,
                target=target,
                ip=ip,
                details=details,
            )
