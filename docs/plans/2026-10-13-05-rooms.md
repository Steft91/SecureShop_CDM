# 05 · Servicio `rooms` — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Gestión de salas por el admin y salas de cada cuenta (el gateway ya verificó la sesión de usuario):
- **canal general** (rev. 4): existe desde la instalación, todas las cuentas lo ven y entran sin clave; el admin cambia su modo y su cupo, pero no lo elimina;
- **unirse** a una sala del admin con código + clave crea una **membresía permanente**: después la sala aparece en la lista de la cuenta y se entra sin volver a pedir la clave, hasta que el admin la elimine o la cuenta la abandone;
- clave de sala con Argon2id (R1) y rate limit al unirse por usuario, IP, dispositivo y sala (R8);
- **entrar** a conversar: una sola sesión de sala activa por usuario, atada al dispositivo; desde otro dispositivo se rechaza y desde el mismo la nueva reemplaza a la anterior (R13); cupo de conectados atómico (R15).

El alias en el chat es el de la cuenta (`X-User-Name`, elegido en el primer ingreso): el cliente no lo manda.

También verifica sesiones para el gateway y lleva el ciclo de vida de cada sesión: salida, fin pedido por `chat`, heartbeat y barrido.

**Architecture:** FastAPI sobre `create_base_app`, con tres routers:
- `admin.py`: CRUD con `require_admin`.
- `join.py`: `/api/device` con `require_gateway`; `GET /api/rooms`, `POST /api/rooms/join` y `DELETE /api/rooms/{id}/membership` con `require_user`.
- `enter.py`: `POST /api/rooms/{id}/enter` con `require_user` (solo miembros, o el general).
- `sessions.py`: `verify-session` para el gateway; `leave` con `require_session`; `end` y `heartbeat` solo para `chat`; fin de las sesiones de una cuenta que cerró sesión, solo para `auth`; barrido en segundo plano; ocupación para las métricas.

Al entrar, en la **misma transacción** se termina la sesión anterior del mismo dispositivo (si la hay), se toma el cupo con `UPDATE … WHERE current < max` y se inserta la sesión, con el índice único parcial por usuario como respaldo. Para cerrar WebSockets se pide a `chat` un `kick`; si falla, el heartbeat reconcilia.

**Tech Stack:** Python 3.13, FastAPI, SQLAlchemy async, Alembic, argon2-cffi (vía `shared.passwords`), httpx, pytest.

**Spec:** `docs/specs/2026-10-12-chat-seguro-design.md` §7.2, §8, §9.3–§9.4, §14, §16.4 (R1, R5, R8, R11, R13, R15, R16).

```python
"""Validación de entrada de rooms (spec §8.1)."""

import re
from typing import Literal

from pydantic import BaseModel, Field, field_validator

CODE_RE = re.compile(r"[a-z0-9_-]{3,32}")


def normalize_code(value: str) -> str:
    value = value.strip().lower()
    if not CODE_RE.fullmatch(value):
        raise ValueError("código inválido")
    return value


class RoomIn(BaseModel):
    code: str
    key: str = Field(min_length=12, max_length=128)
    max_participants: int = Field(ge=2, le=50)
    mode: Literal["text", "media", "both"]

    # Ojo: los validadores NO pueden empezar con "_" (Pydantic los trataría como atributos privados).
    @field_validator("code")
    @classmethod
    def check_code(cls, value: str) -> str:
        return normalize_code(value)


class RoomPatch(BaseModel):
    key: str | None = Field(default=None, min_length=12, max_length=128)
    max_participants: int | None = Field(default=None, ge=2, le=50)
    mode: Literal["text", "media", "both"] | None = None


class JoinIn(BaseModel):
    """El alias no viene del cliente: es el de la cuenta (cabecera X-User-Name del gateway)."""

    code: str
    key: str = Field(min_length=1, max_length=128)

    @field_validator("code")
    @classmethod
    def check_code(cls, value: str) -> str:
        return normalize_code(value)
```

```python
"""Servicio rooms (spec §9.3–§9.4)."""

import asyncio
from contextlib import asynccontextmanager, suppress

import httpx
from fastapi import FastAPI
from sqlalchemy.ext.asyncio import AsyncEngine, create_async_engine

from rooms_svc import admin, enter, join, sessions
from rooms_svc.config import Settings, load_settings
from shared.audit_client import AuditClient
from shared.service_auth import ServiceAuth, ServiceClient
from shared.web import create_base_app


@asynccontextmanager
async def lifespan(app: FastAPI):
    sweeper = asyncio.create_task(sessions.sweeper_loop(app))
    try:
        yield
    finally:
        sweeper.cancel()
        with suppress(asyncio.CancelledError):
            await sweeper


def create_app(
    settings: Settings | None = None,
    *,
    engine: AsyncEngine | None = None,
    service_auth: ServiceAuth | None = None,
    transport: httpx.AsyncBaseTransport | None = None,
) -> FastAPI:
    settings = settings or load_settings()
    service_auth = service_auth or ServiceAuth.from_secrets("rooms", needs_token=True)
    peers = ServiceClient(
        service_auth, {"audit": settings.audit_url, "chat": settings.chat_url}, transport
    )
    app = create_base_app(
        title="rooms",
        settings=settings,
        engine=engine or create_async_engine(settings.database_url, pool_pre_ping=True),
        service_auth=service_auth,
        audit=AuditClient(peers),
        lifespan=lifespan,
    )
    app.state.peers = peers
    for module in (admin, join, enter, sessions):
        app.include_router(module.router)
    return app
```

```python
@dataclass
class Guest:
    """Una cuenta en un navegador: su cookie de dispositivo y, si entró a una sala, su sesión."""

    device: str
    alias: str = "Ana"
    user_id: str = field(default_factory=lambda: str(uuid4()))
    token: str | None = None
    session_id: str | None = None
    room_id: str | None = None

    def cookies(self) -> dict:
        cookies = {DEVICE_COOKIE: self.device}
        if self.token:
            cookies[ROOM_COOKIE] = self.token
        return cookies

    def headers(self) -> dict:
        """Lo que el gateway agrega tras verify-user (X-User-Id y X-User-Name)."""
        return testing.user_headers(self.user_id, self.alias)


def send(client, method: str, url: str, guest: Guest | None = None, **kwargs):
    """Petición con las cookies explícitas del invitado (el jar del TestClient queda vacío)."""
    client.cookies.clear()
    headers = {
        **kwargs.pop("headers", {}),
        **(testing.cookie_header(guest.cookies()) if guest else {}),
    }
    response = client.request(method, url, headers=headers, **kwargs)
    client.cookies.clear()
    return response


def new_guest(client, alias: str = "Ana") -> Guest:
    r = send(client, "GET", "/api/device")
    assert r.status_code == 204, r.text
    return Guest(device=r.cookies[DEVICE_COOKIE], alias=alias)


def join(client, guest: Guest, code="sala-1", key=KEY):
    """Unirse (membresía) con código y clave, como lo ve rooms tras el verify-user del gateway."""
    return send(
        client,
        "POST",
        "/api/rooms/join",
        guest,
        json={"code": code, "key": key},
        headers=guest.headers(),
    )


def enter(client, guest: Guest, room_id: str):
    r = send(client, "POST", f"/api/rooms/{room_id}/enter", guest, headers=guest.headers())
    if r.status_code == 200:
        guest.token = r.cookies[ROOM_COOKIE]
        guest.session_id = r.json()["session_id"]
        guest.room_id = r.json()["room"]["id"]
    return r


def join_and_enter(client, guest: Guest, code="sala-1", key=KEY):
    r = join(client, guest, code, key)
    assert r.status_code == 200, r.text
    return enter(client, guest, r.json()["room"]["id"])
```

```python
"""Ciclo de vida de las sesiones de sala (spec §9.4)."""

import asyncio
import logging

import httpx
from fastapi import APIRouter
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncConnection

router = APIRouter()
log = logging.getLogger("rooms.sessions")


async def end_session(conn: AsyncConnection, session_id: str, reason: str) -> bool:
    """Idempotente: solo la primera llamada termina la sesión y libera el cupo."""
    room_id = (
        await conn.execute(
            text(
                "UPDATE room_sessions SET ended_at = now(), end_reason = :reason "
                "WHERE id = :id AND ended_at IS NULL RETURNING room_id"
            ),
            {"id": session_id, "reason": reason},
        )
    ).scalar()
    if room_id is None:
        return False
    await conn.execute(
        text(
            "UPDATE rooms SET current_participants = current_participants - 1 "
            "WHERE id = :room AND current_participants > 0"
        ),
        {"room": room_id},
    )
    return True


async def kick(app, session_id: str, code: int) -> None:
    """Pide a chat cerrar el WS de la sesión. Si no responde, el heartbeat lo reconcilia (spec §9.4)."""
    try:
        await app.state.peers.post("chat", f"/internal/sessions/{session_id}/kick", json={"code": code})
    except httpx.HTTPError:
        log.warning("chat no respondió al kick de %s; lo reconciliará el heartbeat", session_id)


async def sweeper_loop(app) -> None:
    while True:
        await asyncio.sleep(app.state.settings.sweep_interval_s)
```

```python
"""CRUD de salas para el admin (spec §8.1; riesgos 1, 5 y 8). El canal general no se elimina ni lleva clave."""

from uuid import UUID, uuid4

from fastapi import APIRouter, Depends, HTTPException, Request
from sqlalchemy import text
from sqlalchemy.exc import IntegrityError

from rooms_svc.schemas import RoomIn, RoomPatch
from rooms_svc.sessions import kick
from shared.identity import AdminIdentity, require_admin
from shared.passwords import hash_secret_async, strong_enough

router = APIRouter()
WEAK_KEY = "clave de sala débil: ≥ 12 caracteres, 3 tipos y distinta del código"


@router.get("/api/admin/rooms")
async def list_rooms(request: Request, admin: AdminIdentity = Depends(require_admin)) -> list[dict]:
    async with request.app.state.engine.connect() as conn:
        result = await conn.execute(
            text(
                "SELECT id::text AS id, code, mode, max_participants, current_participants, active, "
                "is_general AS general, created_at FROM rooms ORDER BY is_general DESC, created_at DESC"
            )
        )
        return [dict(row) for row in result.mappings()]


@router.post("/api/admin/rooms", status_code=201)
async def create_room(
    body: RoomIn, request: Request, admin: AdminIdentity = Depends(require_admin)
) -> dict:
    if not strong_enough(body.key, forbid=body.code):
        raise HTTPException(422, WEAK_KEY)
    key_hash = await hash_secret_async(body.key)
    room_id = str(uuid4())
    try:
        async with request.app.state.engine.begin() as conn:
            await conn.execute(
                text(
                    "INSERT INTO rooms (id, code, key_hash, mode, max_participants, created_by) "
                    "VALUES (:id, :code, :hash, :mode, :max, :admin)"
                ),
                {
                    "id": room_id,
                    "code": body.code,
                    "hash": key_hash,
                    "mode": body.mode,
                    "max": body.max_participants,
                    "admin": admin.admin_id,
                },
            )
            await request.app.state.audit.emit(
                "info",
                "room.create",
                actor=f"admin:{admin.username}",
                target=body.code,
                details={"mode": body.mode, "max_participants": body.max_participants},
                required=True,
            )
    except IntegrityError:
        raise HTTPException(409, "código en uso") from None
    return {"id": room_id, "code": body.code}


@router.patch("/api/admin/rooms/{room_id}")
async def update_room(
    room_id: UUID, body: RoomPatch, request: Request, admin: AdminIdentity = Depends(require_admin)
) -> dict:
    engine = request.app.state.engine
    async with engine.connect() as conn:
        room = (
            await conn.execute(
                text("SELECT code, is_general FROM rooms WHERE id = :id AND active"),
                {"id": str(room_id)},
            )
        ).first()
    if room is None:
        raise HTTPException(404, "sala no encontrada")
    code = room.code
    if room.is_general and body.key is not None:
        raise HTTPException(422, "el canal general no tiene clave")
    changes, params, fields = [], {"id": str(room_id)}, []
    if body.key is not None:
        if not strong_enough(body.key, forbid=code):
            raise HTTPException(422, WEAK_KEY)
        changes.append("key_hash = :key_hash")
        params["key_hash"] = await hash_secret_async(body.key)
        fields.append("key")
    if body.max_participants is not None:
        changes.append("max_participants = :max")
        params["max"] = body.max_participants
        fields.append("max_participants")
    if body.mode is not None:
        changes.append("mode = :mode")
        params["mode"] = body.mode
        fields.append("mode")
    if not changes:
        raise HTTPException(422, "sin cambios")
    # Solo fragmentos constantes de esta función: ningún dato del usuario entra al texto SQL.
    statement = f"UPDATE rooms SET {', '.join(changes)} WHERE id = :id"  # noqa: S608  # nosec B608
    try:
        async with engine.begin() as conn:
            await conn.execute(text(statement), params)
            await request.app.state.audit.emit(
                "info",
                "room.update",
                actor=f"admin:{admin.username}",
                target=code,
                details={"fields": fields},
                required=True,
            )
    except IntegrityError:
        raise HTTPException(
            409, "el cupo no puede ser menor que los participantes actuales"
        ) from None
    return {"id": str(room_id), "code": code}


@router.delete("/api/admin/rooms/{room_id}", status_code=204)
async def delete_room(
    room_id: UUID, request: Request, admin: AdminIdentity = Depends(require_admin)
) -> None:
    ended: list[str] = []
    async with request.app.state.engine.begin() as conn:
        general = (
            await conn.execute(
                text("SELECT is_general FROM rooms WHERE id = :id"), {"id": str(room_id)}
            )
        ).scalar()
        if general:
            raise HTTPException(409, "el canal general no se puede eliminar")
        code = (
            await conn.execute(
                text(
                    "UPDATE rooms SET active = false, current_participants = 0 "
                    "WHERE id = :id AND active RETURNING code"
                ),
                {"id": str(room_id)},
            )
        ).scalar()
        if code is not None:
            result = await conn.execute(
                text(
                    "UPDATE room_sessions SET ended_at = now(), end_reason = 'room_deleted' "
                    "WHERE room_id = :id AND ended_at IS NULL RETURNING id::text"
                ),
                {"id": str(room_id)},
            )
            ended = list(result.scalars())
            await request.app.state.audit.emit(
                "info",
                "room.delete",
                actor=f"admin:{admin.username}",
                target=code,
                details={"sessions": len(ended)},
                required=True,
            )
    if code is None:
        raise HTTPException(404, "sala no encontrada")
    for session_id in ended:
        await kick(request.app, session_id, 4401)
```

```python
class Refused(Exception):
    """Corta la transacción del ingreso (rollback del cupo y del reemplazo)."""

    def __init__(self, outcome: str) -> None:
        self.outcome = outcome


async def release_previous(conn, user_id: str, device: bytes, room_id: str) -> str | None:
    """Termina la sesión de sala anterior del MISMO dispositivo ANTES de tomar cupo; desde otro
    dispositivo se rechaza el ingreso nuevo (R13). Devuelve la sesión reemplazada."""
    previous = (
        await conn.execute(
            text(
                "SELECT id::text AS id, room_id::text AS room_id, device_hash FROM room_sessions "
                "WHERE user_id = :user AND ended_at IS NULL FOR UPDATE"
            ),
            {"user": user_id},
        )
    ).first()
    if previous is None:
        return None
    if bytes(previous.device_hash) != device:
        raise Refused("device_conflict")
    # Las salas se bloquean en orden de id para que dos cambios cruzados no se bloqueen mutuamente.
    await conn.execute(
        text("SELECT id FROM rooms WHERE id = ANY(CAST(:ids AS uuid[])) ORDER BY id FOR UPDATE"),
        {"ids": sorted({previous.room_id, room_id})},
    )
    await end_session(conn, previous.id, "replaced")
    return previous.id

        )
```

```python
"""Ciclo de vida de las sesiones de sala (spec §9.4)."""

import asyncio
import logging
from typing import Literal
from uuid import UUID

import httpx
from fastapi import APIRouter, Depends, HTTPException, Request, Response
from pydantic import BaseModel, Field
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncConnection

from shared.crypto import sha256, token_hash
from shared.identity import SessionIdentity, client_ip, encode_alias, require_session
from shared.service_auth import allow
from shared.web import DEVICE_COOKIE, ROOM_COOKIE, clear_cookie

router = APIRouter()
log = logging.getLogger("rooms.sessions")


async def end_session(conn: AsyncConnection, session_id: str, reason: str) -> bool:
    """Idempotente: solo la primera llamada termina la sesión y libera el cupo."""
    room_id = (
        await conn.execute(
            text(
                "UPDATE room_sessions SET ended_at = now(), end_reason = :reason "
                "WHERE id = :id AND ended_at IS NULL RETURNING room_id"
            ),
            {"id": session_id, "reason": reason},
        )
    ).scalar()
    if room_id is None:
        return False
    await conn.execute(
        text(
            "UPDATE rooms SET current_participants = current_participants - 1 "
            "WHERE id = :room AND current_participants > 0"
        ),
        {"room": room_id},
    )
    return True


async def kick(app, session_id: str, code: int) -> None:
    """Pide a chat cerrar el WS de la sesión. Si no responde, el heartbeat lo reconcilia (spec §9.4)."""
    try:
        await app.state.peers.post(
            "chat", f"/internal/sessions/{session_id}/kick", json={"code": code}
        )
    except httpx.HTTPError:
        log.warning("chat no respondió al kick de %s; lo reconciliará el heartbeat", session_id)


@router.get("/internal/verify-session", status_code=204, dependencies=[Depends(allow("gateway"))])
async def verify_session(request: Request, response: Response) -> None:
    """Para el auth_request del gateway: sesión activa + misma cookie de dispositivo (R13)."""
    token, device = request.cookies.get(ROOM_COOKIE), request.cookies.get(DEVICE_COOKIE)
    row = None
    if token and device:
        async with request.app.state.engine.connect() as conn:
            row = (
                await conn.execute(
                    text(
                        "SELECT s.id::text AS session_id, s.room_id::text AS room_id, "
                        "s.user_id::text AS user_id, s.alias, r.mode "
                        "FROM room_sessions s JOIN rooms r ON r.id = s.room_id "
                        "WHERE s.token_hash = :token AND s.device_hash = :device "
                        "AND s.ended_at IS NULL AND r.active"
                    ),
                    {"token": token_hash(token), "device": sha256(device.encode())},
                )
            ).first()
    if row is None:
        ip = client_ip(request)
        await request.app.state.audit.emit(
            "warning",
            "gateway.reject.session",
            actor=f"ip:{ip}",
            target=request.headers.get("x-original-uri"),
            ip=ip,
        )
        raise HTTPException(401, "no autenticado")
    response.headers["X-Session-Id"] = row.session_id
    response.headers["X-Room-Id"] = row.room_id
    response.headers["X-User-Id"] = row.user_id
    response.headers["X-Alias"] = encode_alias(row.alias)
    response.headers["X-Room-Mode"] = row.mode


class EndIn(BaseModel):
    reason: Literal["closed", "violation", "idle"]


class HeartbeatIn(BaseModel):
    session_ids: list[UUID] = Field(default_factory=list, max_length=10_000)


async def end_and_kick(app, session_id: str, reason: str, code: int | None) -> bool:
    async with app.state.engine.begin() as conn:
        ended = await end_session(conn, session_id, reason)
    if ended:
        await app.state.audit.emit(
            "info", "session.end", actor=f"session:{session_id}", details={"reason": reason}
        )
        if code is not None:
            await kick(app, session_id, code)
    return ended


@router.post("/api/rooms/leave", status_code=204)
async def leave(
    request: Request, response: Response, who: SessionIdentity = Depends(require_session)
) -> None:
    """Salir de la conversación libera el cupo; la membresía de la sala se conserva."""
    await end_and_kick(request.app, who.session_id, "leave", 1000)
    clear_cookie(response, ROOM_COOKIE)


@router.post(
    "/internal/sessions/{session_id}/end", status_code=204, dependencies=[Depends(allow("chat"))]
)
async def end(session_id: UUID, body: EndIn, request: Request) -> None:
    await end_and_kick(request.app, str(session_id), body.reason, None)


@router.post(
    "/internal/users/{user_id}/sessions/end", status_code=204, dependencies=[Depends(allow("auth"))]
)
async def end_user_sessions(user_id: UUID, request: Request) -> None:
    """auth avisa que la cuenta cerró sesión: se corta su sesión de sala y su WebSocket (RS-07)."""
    async with request.app.state.engine.connect() as conn:
        active = (
            await conn.execute(
                text(
                    "SELECT id::text FROM room_sessions WHERE user_id = :user AND ended_at IS NULL"
                ),
                {"user": str(user_id)},
            )
        ).scalars()
        sessions = list(active)
    for session_id in sessions:
        await end_and_kick(request.app, session_id, "logout", 4401)


@router.post("/internal/sessions/heartbeat", dependencies=[Depends(allow("chat"))])
async def heartbeat(body: HeartbeatIn, request: Request) -> dict:
    """chat reporta sus conexiones vivas; se devuelven las que rooms ya terminó (reconciliación)."""
    ids = [str(session_id) for session_id in body.session_ids]
    if not ids:
        return {"ended": []}
    async with request.app.state.engine.begin() as conn:
        result = await conn.execute(
            text(
                "UPDATE room_sessions SET last_seen = now() "
                "WHERE id = ANY(CAST(:ids AS uuid[])) AND ended_at IS NULL RETURNING id::text"
            ),
            {"ids": ids},
        )
        alive = set(result.scalars())
    return {"ended": [session_id for session_id in ids if session_id not in alive]}


async def sweep(app) -> list[str]:
    """Termina las sesiones sin heartbeat de chat por más del período de gracia (spec §9.4)."""
    settings = app.state.settings
    async with app.state.engine.begin() as conn:
        result = await conn.execute(
            text(
                "SELECT id::text FROM room_sessions WHERE ended_at IS NULL "
                "AND last_seen < now() - make_interval(secs => :grace) FOR UPDATE SKIP LOCKED"
            ),
            {"grace": settings.room_session_grace_s},
        )
        stale = list(result.scalars())
        for session_id in stale:
            await end_session(conn, session_id, "grace")
        await conn.execute(
            text("DELETE FROM failed_attempts WHERE created_at < now() - interval '1 day'")
        )
    for session_id in stale:
        await app.state.audit.emit(
            "info", "session.end", actor=f"session:{session_id}", details={"reason": "grace"}
        )
        await kick(app, session_id, 4401)
    return stale


async def sweeper_loop(app) -> None:
    while True:
        await asyncio.sleep(app.state.settings.sweep_interval_s)
        try:
            await sweep(app)
        except Exception:
            log.exception("falló el barrido de sesiones")


async def occupancy(app) -> dict:
    """Métrica propia de rooms para el panel (RNF-18): ocupación de cada sala activa."""
    async with app.state.engine.connect() as conn:
        result = await conn.execute(
            text(
                "SELECT code, current_participants AS connected, max_participants AS max "
                "FROM rooms WHERE active ORDER BY is_general DESC, code"
            )
        )
        return {"rooms": [dict(row) for row in result.mappings()]}
```

```python
"""Servicio rooms (spec §9.3–§9.4)."""

import asyncio
from contextlib import asynccontextmanager, suppress

import httpx
from fastapi import FastAPI
from sqlalchemy.ext.asyncio import AsyncEngine, create_async_engine

from rooms_svc import admin, enter, join, sessions
from rooms_svc.config import Settings, load_settings
from shared.audit_client import AuditClient
from shared.service_auth import ServiceAuth, ServiceClient
from shared.web import create_base_app


@asynccontextmanager
async def lifespan(app: FastAPI):
    sweeper = asyncio.create_task(sessions.sweeper_loop(app))
    try:
        yield
    finally:
        sweeper.cancel()
        with suppress(asyncio.CancelledError):
            await sweeper


def create_app(
    settings: Settings | None = None,
    *,
    engine: AsyncEngine | None = None,
    service_auth: ServiceAuth | None = None,
    transport: httpx.AsyncBaseTransport | None = None,
) -> FastAPI:
    settings = settings or load_settings()
    service_auth = service_auth or ServiceAuth.from_secrets("rooms", needs_token=True)
    peers = ServiceClient(
        service_auth, {"audit": settings.audit_url, "chat": settings.chat_url}, transport
    )
    app = create_base_app(
        title="rooms",
        settings=settings,
        engine=engine or create_async_engine(settings.database_url, pool_pre_ping=True),
        service_auth=service_auth,
        audit=AuditClient(peers),
        lifespan=lifespan,
    )
    app.state.peers = peers
    app.state.extra_metrics = sessions.occupancy
    for module in (admin, join, enter, sessions):
        app.include_router(module.router)
    return app
```
