# 07 · Servicio `media` — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Recibir imágenes en las salas que lo permiten y procesarlas así:
1. validar tipo real, integridad y tamaño (R9);
2. escanear con ClamAV (R9);
3. detectar esteganografía con chi², RS y estructura (R6);
4. aplicar la política: cuarentena si es sospechosa, rechazo si es inválida o maliciosa, sanitización y publicación en la sala si es limpia.

Las descargas exigen URL firmada y pertenencia a la sala (R12). El admin revisa la cuarentena.

**Architecture:**
- `formats.py` (punto de extensión `FORMATS`) une firma, validación con Pillow, `analyze` y `sanitize` por formato.
- `png.py`/`jpeg.py` implementan el análisis y la sanitización; `stego.py`, los estadísticos.
- `clamav.py` es un cliente INSTREAM con la stdlib.
- `pipeline.py` orquesta: el análisis corre en un `ProcessPoolExecutor` con timeout, se reconocen por `clean_sha256` las salidas propias y se publica en `chat`.
- `routes.py` contiene la subida y la descarga; `admin.py`, la cuarentena.

**Tech Stack:** Python 3.13, FastAPI, Pillow, numpy, scipy, asyncio, SQLAlchemy async, ClamAV 1.4, pytest.

**Spec:** `docs/specs/2026-10-12-chat-seguro-design.md` §7.4, §8, §9.5–§9.7, §12, §16.4 (R6, R9, R11, R12, R16).

## Global Constraints

Las del índice (`2026-10-13-00-indice.md`). En particular:
- Subida ≤ 5 MB; imagen ≤ 20 MP.
- ClamAV falla cerrado (503).
- Todo archivo publicado pasó por la sanitización.
- Descargas solo con firma HMAC válida, de la misma sesión y de la misma sala.
- `text()` con parámetros nombrados.
- Commits en español, sin atribución.

from collections.abc import Awaitable, Callable

import httpx
from fastapi import FastAPI
from sqlalchemy.ext.asyncio import AsyncEngine, create_async_engine

from media_svc import admin, routes
from media_svc.config import Settings, load_settings
from shared.audit_client import AuditClient
from shared.service_auth import ServiceAuth, ServiceClient
from shared.web import create_base_app

Scanner = Callable[[bytes], Awaitable[str | None]]


def create_app(
    settings: Settings | None = None,
    *,
    engine: AsyncEngine | None = None,
    service_auth: ServiceAuth | None = None,
    transport: httpx.AsyncBaseTransport | None = None,
    scanner: Scanner | None = None,
) -> FastAPI:
    settings = settings or load_settings()
    service_auth = service_auth or ServiceAuth.from_secrets("media", needs_token=True)
    peers = ServiceClient(service_auth, {"audit": settings.audit_url, "chat": settings.chat_url}, transport)
    app = create_base_app(
        title="media",
        settings=settings,
        engine=engine or create_async_engine(settings.database_url, pool_pre_ping=True),
        service_auth=service_auth,
        audit=AuditClient(peers),
    )
    app.state.peers = peers
    app.state.scanner = scanner  # MED-3 pone ClamAV por defecto
    for module in (routes, admin):
        app.include_router(module.router)
    return app

```python
"""Estadísticos de estegoanálisis LSB (spec §12.2). Funciones puras sobre arrays numpy."""

import math

import numpy as np
from scipy.stats import chi2

MASK = np.array([0, 1, 1, 0], dtype=bool)


def finite(value: float) -> float | None:
    """jsonb no admite NaN ni infinitos: se guardan como null."""
    value = float(value)
    return value if math.isfinite(value) else None


def chi2_pvalue(values: np.ndarray) -> float:
    """Ataque chi² de Westfeld–Pfitzmann: p ≈ 1 si los pares (2k, 2k+1) están igualados."""
    hist = np.bincount(values.ravel(), minlength=256).astype(np.float64)
    even, odd = hist[0::2], hist[1::2]
    expected = (even + odd) / 2
    mask = expected > 4
    dof = int(mask.sum()) - 1
    if dof < 1:
        return 0.0
    statistic = float((((even - expected) ** 2)[mask] / expected[mask]).sum())
    return float(chi2.sf(statistic, dof))


def chi2_scan(pixels: np.ndarray, min_pixels: int = 1000) -> float:
    """Máximo p sobre prefijos de 10/25/50/100 % por canal (el embedding secuencial deja el prefijo con p ≈ 1)."""
    flat = pixels.reshape(-1, pixels.shape[-1]) if pixels.ndim == 3 else pixels.reshape(-1, 1)
    best = 0.0
    for fraction in (0.1, 0.25, 0.5, 1.0):
        count = int(len(flat) * fraction)
        if count < min_pixels and fraction < 1.0:
            continue
        for channel in range(flat.shape[1]):
            best = max(best, chi2_pvalue(flat[:count, channel]))
    return best


def _groups(channel: np.ndarray) -> np.ndarray:
    """Bloques 2×2 aplanados en zigzag: (0,0) (0,1) (1,1) (1,0)."""
    h, w = channel.shape[0] // 2 * 2, channel.shape[1] // 2 * 2
    blocks = channel[:h, :w].astype(np.int16).reshape(h // 2, 2, w // 2, 2).transpose(0, 2, 1, 3)
    return blocks.reshape(-1, 4)[:, [0, 1, 3, 2]]


def _variation(groups: np.ndarray) -> np.ndarray:
    return np.abs(np.diff(groups, axis=1)).sum(axis=1)


def _rs_counts(groups: np.ndarray) -> tuple[float, float, float, float]:
    base = _variation(groups)
    positive, negative = groups.copy(), groups.copy()
    positive[:, MASK] = positive[:, MASK] ^ 1  # F1: 0↔1, 2↔3, …
    negative[:, MASK] = ((negative[:, MASK] + 1) ^ 1) - 1  # F−1: −1↔0, 1↔2, …
    fp, fn = _variation(positive), _variation(negative)
    return (fp > base).mean(), (fp < base).mean(), (fn > base).mean(), (fn < base).mean()


def rs_estimate(channel: np.ndarray) -> float:
    """Análisis RS (Fridrich, Goljan, Du 2001): tasa de embedding estimada p̂ en [0, 1].

    Degenera cerca del 100 % de embedding; ahí lo cubre el chi².
    """
    groups = _groups(channel)
    if len(groups) == 0:
        return 0.0
    rm, sm, rnm, snm = _rs_counts(groups)
    r1, s1, rn1, sn1 = _rs_counts(groups ^ 1)
    d0, d1, dn0, dn1 = rm - sm, r1 - s1, rnm - snm, rn1 - sn1
    a, b, c = 2 * (d1 + d0), dn0 - dn1 - d1 - 3 * d0, d0 - dn0
    if abs(a) < 1e-12:
        if abs(b) < 1e-12:
            return 0.0
        x = -c / b
    else:
        discriminant = b * b - 4 * a * c
        if discriminant < 0:
            return 0.0
        roots = ((-b + math.sqrt(discriminant)) / (2 * a), (-b - math.sqrt(discriminant)) / (2 * a))
        x = min(roots, key=abs)
    if abs(x - 0.5) < 1e-12:
        return 0.0
    return float(min(max(x / (x - 0.5), 0.0), 1.0))


def lsb_entropy(pixels: np.ndarray, block: int = 64) -> tuple[float, float]:
    """Entropía del plano LSB por bloques: (media, fracción de bloques con H ≥ 0.999). Métrica de apoyo."""
    planes = pixels[..., None] if pixels.ndim == 2 else pixels
    h, w = planes.shape[:2]
    bh, bw = min(block, h), min(block, w)
    nh, nw = h // bh, w // bw
    bits = (planes[: nh * bh, : nw * bw] & 1).astype(np.float64)
    p = bits.reshape(nh, bh, nw, bw, -1).mean(axis=(1, 3))
    with np.errstate(divide="ignore", invalid="ignore"):
        entropy = -(np.where(p > 0, p * np.log2(p), 0) + np.where(p < 1, (1 - p) * np.log2(1 - p), 0))
    return float(entropy.mean()), float((entropy >= 0.999).mean())
```

"""PNG: estructura, estegoanálisis en píxeles y sanitización (spec §12.2, §12.5).

`analyze` y `sanitize` son funciones de módulo: las ejecuta un ProcessPoolExecutor.
"""

import io

import numpy as np
from PIL import Image

from media_svc import stego

KNOWN_CHUNKS = {
    b"IHDR", b"PLTE", b"IDAT", b"IEND", b"tRNS", b"cHRM", b"gAMA", b"iCCP", b"sBIT", b"sRGB",
    b"cICP", b"mDCV", b"cLLI", b"bKGD", b"hIST", b"pHYs", b"sPLT", b"eXIf", b"tIME", b"acTL",
    b"fcTL", b"fdAT", b"tEXt", b"zTXt", b"iTXt",
}
TEXT_CHUNKS = {b"tEXt", b"zTXt", b"iTXt"}


def structure(data: bytes) -> dict:
    position, text_bytes, unknown, trailing = 8, 0, [], 0
    while position + 12 <= len(data):
        length = int.from_bytes(data[position : position + 4], "big")
        ctype = data[position + 4 : position + 8]
        end = position + 12 + length
        if ctype in TEXT_CHUNKS:
            text_bytes += length
        elif ctype not in KNOWN_CHUNKS:
            unknown.append(ctype.decode("latin-1"))
        position = end
        if ctype == b"IEND":
            trailing = max(0, len(data) - end)
            break
    return {"trailing_bytes": trailing, "text_bytes": text_bytes, "unknown_chunks": unknown}


def analyze(data: bytes, cfg: dict) -> dict:
    found = structure(data)
    with Image.open(io.BytesIO(data), formats=["PNG"]) as image:
        rgb = np.asarray(image.convert("RGB"))
    chi2_p = stego.chi2_scan(rgb)
    rs_rate = max(stego.rs_estimate(rgb[..., channel]) for channel in range(3))
    entropy_mean, entropy_high = stego.lsb_entropy(rgb)
    reasons = []
    if found["trailing_bytes"]:
        reasons.append("datos después de IEND")
    if found["text_bytes"] > 1024:
        reasons.append("chunks de texto grandes")
    if found["unknown_chunks"]:
        reasons.append("chunks desconocidos")
    if chi2_p >= cfg["chi2_p"]:
        reasons.append("chi² del plano LSB")
    if rs_rate >= cfg["rs_rate"]:
        reasons.append("análisis RS")
    trigger = cfg.get("entropy_trigger")
    if trigger is not None and entropy_high >= trigger:
        reasons.append("entropía del plano LSB")
    metrics = {
        **found,
        "chi2_p": stego.finite(chi2_p),
        "rs_rate": stego.finite(rs_rate),
        "lsb_entropy_mean": stego.finite(entropy_mean),
        "lsb_entropy_high_fraction": stego.finite(entropy_high),
    }
    return {"suspicious": bool(reasons), "reasons": reasons, "metrics": metrics}


def sanitize(data: bytes) -> bytes:
    """XOR del plano LSB con bits aleatorios (destruye LSB-1 aunque no se haya detectado) y
    recodificación sin chunks auxiliares. Recodificar sin tocar píxeles NO basta: PNG es sin pérdida."""
    with Image.open(io.BytesIO(data), formats=["PNG"]) as image:
        has_alpha = image.mode in ("RGBA", "LA", "PA") or "transparency" in image.info
        converted = image.convert("RGBA" if has_alpha else "RGB")
    pixels = np.array(converted)
    pixels ^= np.random.default_rng().integers(0, 2, size=pixels.shape, dtype=np.uint8)
    out = io.BytesIO()
    Image.fromarray(pixels).save(out, format="PNG", optimize=True)
    return out.getvalue()

"""JPEG: estructura y sanitización (spec §12.3, §12.5). El análisis LSB en píxeles no aplica a JPEG."""

import io

from PIL import Image, ImageOps


def structure(data: bytes) -> dict:
    """Recorre los segmentos por su longitud (una miniatura EXIF con su propio FFD9 no confunde)."""
    position, metadata, end, size = 2, 0, None, len(data)
    while position + 1 < size:
        if data[position] != 0xFF:
            break
        marker = data[position + 1]
        if marker == 0xFF:  # relleno
            position += 1
            continue
        if marker == 0xD9:  # EOI
            end = position + 2
            break
        if 0xD0 <= marker <= 0xD7 or marker == 0x01:  # marcadores sin longitud
            position += 2
            continue
        if position + 4 > size:
            break
        length = int.from_bytes(data[position + 2 : position + 4], "big")
        if 0xE0 <= marker <= 0xEF or marker == 0xFE:  # APPn y COM
            metadata += length
        position += 2 + length
        if marker == 0xDA:  # SOS: saltar los datos de entropía hasta el siguiente marcador real
            while True:
                position = data.find(b"\xff", position)
                if position < 0 or position + 1 >= size:
                    position = size
                    break
                following = data[position + 1]
                if following == 0x00 or 0xD0 <= following <= 0xD7:
                    position += 2
                    continue
                break
    return {"trailing_bytes": size - end if end is not None else 0, "metadata_bytes": metadata}


def analyze(data: bytes, cfg: dict) -> dict:
    found = structure(data)
    reasons = []
    if found["trailing_bytes"] > 16:
        reasons.append("datos después de EOI")
    if found["metadata_bytes"] > 65536:
        reasons.append("metadatos excesivos")
    return {"suspicious": bool(reasons), "reasons": reasons, "metrics": found}


def sanitize(data: bytes) -> bytes:
    """Aplica la orientación EXIF y recodifica en RGB con calidad 90, sin EXIF, ICC ni comentarios."""
    with Image.open(io.BytesIO(data), formats=["JPEG"]) as image:
        rgb = ImageOps.exif_transpose(image).convert("RGB")
    out = io.BytesIO()
    rgb.save(out, format="JPEG", quality=90)
    return out.getvalue()
```

import io
from collections.abc import Callable
from dataclasses import dataclass

from PIL import Image

from media_svc import jpeg, png


@dataclass(frozen=True)
class Format:
    mime: str
    ext: str
    signature: bytes
    pil_format: str
    analyze: Callable[[bytes, dict], dict]  # función de módulo: corre en otro proceso
    sanitize: Callable[[bytes], bytes]


FORMATS = {
    "image/png": Format("image/png", "png", b"\x89PNG\r\n\x1a\n", "PNG", png.analyze, png.sanitize),
    "image/jpeg": Format("image/jpeg", "jpg", b"\xff\xd8\xff", "JPEG", jpeg.analyze, jpeg.sanitize),
}


def detect(data: bytes) -> Format | None:
    """Por firma (magic bytes): la extensión y el Content-Type del cliente se ignoran."""
    for fmt in FORMATS.values():
        if data.startswith(fmt.signature):
            return fmt
    return None


def validate(data: bytes, fmt: Format, max_pixels: int) -> None:
    """Lanza si no es una imagen íntegra del formato que indica la firma o si supera el tope de píxeles."""
    # formats=[...] limita a Pillow a un solo decodificador: menos superficie de ataque.
    with Image.open(io.BytesIO(data), formats=[fmt.pil_format]) as image:
        if image.width * image.height > max_pixels:  # antes de decodificar: bombas de descompresión
            raise ValueError("imagen demasiado grande")
        image.verify()
    with Image.open(io.BytesIO(data), formats=[fmt.pil_format]) as image:
        image.load()

"""Cliente mínimo de clamd (protocolo INSTREAM) con la stdlib (spec §9.5, riesgo 9)."""

import asyncio
import struct

CHUNK = 64 * 1024


class ClamAVUnavailable(Exception):
    """clamd no respondió: el pipeline falla cerrado (503)."""


async def scan(host: str, port: int, data: bytes, timeout: float = 30.0) -> str | None:
    """Nombre de la firma si hay malware; None si está limpio."""
    try:
        reader, writer = await asyncio.wait_for(asyncio.open_connection(host, port), timeout)
    except (OSError, TimeoutError) as exc:
        raise ClamAVUnavailable(str(exc)) from exc
    try:
        writer.write(b"zINSTREAM\0")
        for start in range(0, len(data), CHUNK):
            chunk = data[start : start + CHUNK]
            writer.write(struct.pack("!I", len(chunk)) + chunk)
        writer.write(struct.pack("!I", 0))
        await writer.drain()
        raw = await asyncio.wait_for(reader.readuntil(b"\0"), timeout)
    except (OSError, TimeoutError, asyncio.IncompleteReadError) as exc:
        raise ClamAVUnavailable(str(exc)) from exc
    finally:
        writer.close()
    reply = raw.rstrip(b"\0").decode(errors="replace")
    if reply.endswith(" OK"):
        return None
    if reply.endswith(" FOUND"):
        return reply.removeprefix("stream: ").removesuffix(" FOUND")
    raise ClamAVUnavailable(reply)

"""Pipeline de subida (spec §9.5–§9.6; riesgos 6, 9 y 12)."""

import asyncio
import json
import logging
import os
from concurrent.futures import ProcessPoolExecutor
from dataclasses import dataclass
from pathlib import Path
from uuid import uuid4

import httpx
from fastapi import HTTPException
from sqlalchemy import text
from starlette.concurrency import run_in_threadpool

from media_svc import formats
from media_svc.clamav import ClamAVUnavailable
from shared.crypto import sha256
from shared.identity import SessionIdentity

log = logging.getLogger("media.pipeline")
_POOL: ProcessPoolExecutor | None = None

INSERT_FILE = text(
    "INSERT INTO files (id, room_id, session_id, alias, storage_name, mime, size, sha256, "
    "clean_sha256, status, verdict) VALUES (:id, :room, :session, :alias, :name, :mime, :size, "
    ":sha, :clean_sha, :status, CAST(CAST(:verdict AS text) AS jsonb))"
)


def pool() -> ProcessPoolExecutor:
    # ponytail: pool global de 2 procesos. Un análisis que excede el timeout sigue corriendo hasta
    # terminar (un worker no se puede matar); si eso molesta, mover el análisis a un worker externo.
    global _POOL
    if _POOL is None:
        _POOL = ProcessPoolExecutor(max_workers=2)
    return _POOL


@dataclass(frozen=True)
class Result:
    status: str  # clean | quarantined | rejected
    file_id: str | None


def storage_dir(settings, kind: str) -> Path:
    path = settings.media_dir / kind
    path.mkdir(parents=True, exist_ok=True)
    return path


def write_private(path: Path, data: bytes) -> None:
    path.write_bytes(data)
    os.chmod(path, 0o640)


def file_row(file_id, who: SessionIdentity, name, mime, data, digest, clean_sha, status, verdict) -> dict:
    return {
        "id": file_id,
        "room": who.room_id,
        "session": who.session_id,
        "alias": who.alias,
        "name": name,
        "mime": mime,
        "size": len(data),
        "sha": digest,
        "clean_sha": clean_sha,
        "status": status,
        "verdict": json.dumps(verdict, ensure_ascii=False),
    }


async def reject(app, who, data, digest, reason, mime, ip, severity="warning", details=None) -> Result:
    """No se guarda el archivo; queda constancia de hash, tipo y motivo (spec §9.5)."""
    async with app.state.engine.begin() as conn:
        row = file_row(str(uuid4()), who, None, mime, data, digest, None, "rejected", {"reason": reason, **(details or {})})
        await conn.execute(INSERT_FILE, row)
    await app.state.audit.emit(
        severity, f"file.rejected.{reason}", actor=f"session:{who.session_id}", ip=ip,
        details={"sha256": digest.hex(), "mime": mime, **(details or {})},
    )
    return Result("rejected", None)


async def is_own_output(app, digest: bytes) -> bool:
    """¿Son bytes que el propio sistema sanitizó y publicó? (spec §9.5, paso 5)."""
    async with app.state.engine.connect() as conn:
        found = await conn.execute(
            text(
                "SELECT 1 FROM files WHERE clean_sha256 = :sha "
                "AND status IN ('clean', 'released') LIMIT 1"
            ),
            {"sha": digest},
        )
        return found.first() is not None


async def analyze(app, fmt: formats.Format, data: bytes) -> dict:
    settings = app.state.settings
    loop = asyncio.get_running_loop()
    try:
        return await asyncio.wait_for(
            loop.run_in_executor(pool(), fmt.analyze, data, settings.stego_config),
            settings.stego_timeout_s,
        )
    except TimeoutError:
        return {"suspicious": True, "reasons": ["timeout del análisis"], "metrics": {}}


async def publish(app, room_id: str, file_id: str, session_id: str, alias: str) -> None:
    try:
        response = await app.state.peers.post(
            "chat",
            f"/internal/rooms/{room_id}/files",
            json={"file_id": file_id, "session_id": session_id, "alias": alias},
        )
    except httpx.HTTPError:
        response = None
    if response is None or response.status_code != 204:
        log.error("chat no publicó el archivo %s; queda sin URL firmada (spec §9.6)", file_id)
        raise HTTPException(503, "no se pudo publicar el archivo")


async def process(app, who: SessionIdentity, data: bytes, ip: str | None) -> Result:
    settings, audit = app.state.settings, app.state.audit
    digest = sha256(data)
    if len(data) > settings.upload_max_bytes:
        return await reject(app, who, data, digest, "size", "unknown", ip)
    fmt = formats.detect(data)
    if fmt is None:
        return await reject(app, who, data, digest, "type", "unknown", ip)
    try:
        await run_in_threadpool(formats.validate, data, fmt, settings.max_image_pixels)
    except Exception:  # cualquier fallo al decodificar = imagen inválida
        return await reject(app, who, data, digest, "invalid", fmt.mime, ip)
    try:
        signature = await app.state.scanner(data)
    except ClamAVUnavailable:
        await audit.emit("warning", "clamav.unavailable", actor=f"session:{who.session_id}", ip=ip)
        raise HTTPException(503, "análisis no disponible") from None
    if signature:
        return await reject(
            app, who, data, digest, "malware", fmt.mime, ip, severity="critical",
            details={"signature": signature},
        )
    if await is_own_output(app, digest):
        verdict = {"suspicious": False, "reasons": [], "metrics": {}, "known_output": True}
    else:
        verdict = await analyze(app, fmt, data)

    file_id, name = str(uuid4()), uuid4().hex
    if verdict["suspicious"]:
        write_private(storage_dir(settings, "quarantine") / name, data)
        async with app.state.engine.begin() as conn:
            await conn.execute(INSERT_FILE, file_row(file_id, who, name, fmt.mime, data, digest, None, "quarantined", verdict))
        await audit.emit(
            "warning", "file.quarantined", actor=f"session:{who.session_id}", target=file_id, ip=ip,
            details={"reasons": verdict["reasons"], "metrics": verdict["metrics"]},
        )
        return Result("quarantined", file_id)

    clean = await run_in_threadpool(fmt.sanitize, data)
    path = storage_dir(settings, "media") / name
    write_private(path, clean)
    try:
        async with app.state.engine.begin() as conn:
            await conn.execute(
                INSERT_FILE, file_row(file_id, who, name, fmt.mime, data, digest, sha256(clean), "clean", verdict)
            )
            await audit.emit(
                "info", "file.accepted", actor=f"session:{who.session_id}", target=file_id, ip=ip,
                required=True,
            )
    except Exception:
        path.unlink(missing_ok=True)  # sin fila no puede quedar un archivo huérfano en disco
        raise
    await publish(app, who.room_id, file_id, who.session_id, who.alias)
    return Result("clean", file_id)
```

"""Subida y descarga de archivos (spec §9.5–§9.6; riesgos 6, 9 y 12)."""

import time
from uuid import UUID

from fastapi import APIRouter, Depends, HTTPException, Request, UploadFile
from fastapi.responses import FileResponse
from sqlalchemy import text

from media_svc import formats, pipeline
from shared.crypto import verify_url
from shared.identity import SessionIdentity, client_ip, require_session

router = APIRouter()


@router.post("/api/media/files")
async def upload(request: Request, file: UploadFile, who: SessionIdentity = Depends(require_session)) -> dict:
    settings, engine, audit = request.app.state.settings, request.app.state.engine, request.app.state.audit
    ip = client_ip(request)
    if who.mode not in ("media", "both"):
        await audit.emit("warning", "file.rejected.forbidden", actor=f"session:{who.session_id}", ip=ip)
        raise HTTPException(403, "la sala no permite archivos")
    async with engine.connect() as conn:
        recent = (
            await conn.execute(
                text(
                    "SELECT count(*) FROM files "
                    "WHERE session_id = :session AND created_at > now() - interval '1 minute'"
                ),
                {"session": who.session_id},
            )
        ).scalar_one()
    if recent >= settings.upload_per_min:
        raise HTTPException(429, "demasiadas subidas; espera un minuto")
    data = await file.read(settings.upload_max_bytes + 1)
    result = await pipeline.process(request.app, who, data, ip)
    if result.status == "rejected":
        raise HTTPException(422, "archivo no permitido")
    return {"file_id": result.file_id, "status": result.status}


@router.get("/api/media/files/{file_id}")
async def download(
    file_id: UUID,
    request: Request,
    exp: int = 0,
    sig: str = "",
    who: SessionIdentity = Depends(require_session),
):
    settings, fid = request.app.state.settings, str(file_id)
    async with request.app.state.engine.connect() as conn:
        row = (
            await conn.execute(
                text("SELECT room_id::text AS room_id, storage_name, mime, status FROM files WHERE id = :id"),
                {"id": fid},
            )
        ).first()
    allowed = (
        row is not None
        and row.status in ("clean", "released")
        and row.room_id == who.room_id
        and verify_url(settings.url_signing_key, fid, who.session_id, exp, sig, int(time.time()))
    )
    if not allowed:  # 404 genérico: no revela si el archivo existe
        await request.app.state.audit.emit(
            "warning", "file.download.denied", actor=f"session:{who.session_id}", target=fid,
            ip=client_ip(request),
        )
        raise HTTPException(404, "no encontrado")
    return FileResponse(
        settings.media_dir / "media" / row.storage_name,
        media_type=row.mime,
        headers={
            "X-Content-Type-Options": "nosniff",
            "Content-Disposition": f'inline; filename="image.{formats.FORMATS[row.mime].ext}"',
            "Cache-Control": "private, no-store",
        },
    )
