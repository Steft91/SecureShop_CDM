# 03 · Servicio `auth` — Plan de implementación (rev. 4)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Cuentas con registro abierto, roles dinámicos con permisos de un catálogo fijo (RBAC), login con Argon2id, bloqueo y retraso progresivo, TOTP obligatorio para los permisos de admin, un JWT RS256 que apunta a una sesión revocable ligada a un único dispositivo, y el endpoint `verify` que el gateway llama en cada request para obtener la identidad y los permisos **actuales** (riesgos R4, R5, R13 y R16). El secreto TOTP queda cifrado con una KEK rotable (R2).

**Architecture:**
- Rutas públicas en `routes.py` (`/api/auth/*`) y de administración en `admin.py` (`/api/admin/*`); internas en `internal.py` (`/internal/verify` para el gateway y `/internal/sessions/heartbeat` para chat).
- `tokens.Signer` firma y verifica los JWT con el algoritmo fijo `RS256`. El JWT nunca se guarda: la BD solo tiene la sesión (`user_sessions`), que decide revocación, inactividad y dispositivo.
- `sessions.py` aplica la política de dispositivo único: un índice único parcial impide dos sesiones `full` vivas por cuenta, y el login desde otro dispositivo se rechaza.
- Eventos a `audit` vía `AuditClient`: los que otorgan algo son obligatorios; los que quitan acceso no esperan a la auditoría.

**Tech Stack:** Python 3.13, FastAPI, SQLAlchemy async, Alembic, argon2-cffi (vía `shared.passwords`), PyJWT, cryptography, pyotp, qrcode, pytest.

**Spec:** `docs/specs/2026-10-12-chat-seguro-design.md` §7.1, §8.1–§8.2, §9.1, §11, §14.2, §14.4, §16.4 (R2, R4, R5, R11, R13, R16).

## Global Constraints

Las del índice (`2026-10-13-00-indice.md`). En particular:
- `text()` con parámetros nombrados.
- Todas las rutas exigen `X-Service-Token` salvo `/health`.
- Errores `{"error": "..."}` genéricos; cuerpos con `extra="forbid"`.
- Cookies `__Host-session` (JWT) y `__Host-device`, con `Secure`, `HttpOnly` y `SameSite=Strict`.
- Argon2id con ≤ 4 cálculos concurrentes.
- JWT solo con `algorithms=["RS256"]`; claims `iss`, `aud`, `sub`, `sid`, `name`, `stage`, `roles`, `iat`, `exp`.
- Commits en español, sin atribución.

`services/auth/auth_svc/tokens.py`:
```python
"""JWT RS256 que apunta a una sesión del servidor (spec §8.1; riesgo 16).

El algoritmo va fijo al verificar: un token con `alg: none`, o uno firmado con HS256 usando la
clave pública como secreto, no pasa. La revocación no depende del `exp`: la decide la sesión (verify).
"""

import time
from datetime import datetime

import jwt
from cryptography.hazmat.primitives import serialization

REQUIRED_CLAIMS = ["iss", "aud", "sub", "sid", "stage", "iat", "exp"]


class Signer:
    def __init__(self, private_pem: str, issuer: str, audience: str) -> None:
        self._private = serialization.load_pem_private_key(private_pem.encode(), password=None)
        self._public = self._private.public_key()
        self.issuer, self.audience = issuer, audience

    def issue(
        self,
        *,
        user_id: str,
        username: str,
        session_id: str,
        stage: str,
        roles: list[str],
        expires_at: datetime,
    ) -> str:
        claims = {
            "iss": self.issuer,
            "aud": self.audience,
            "sub": user_id,
            "sid": session_id,
            "name": username,
            "stage": stage,
            "roles": roles,  # solo informativo: la autorización usa los permisos de la BD
            "iat": int(time.time()),
            "exp": int(expires_at.timestamp()),
        }
        return jwt.encode(claims, self._private, algorithm="RS256")

    def read(self, token: str) -> dict | None:
        """Claims de un JWT con firma, emisor, audiencia y vencimiento válidos; None en otro caso."""
        try:
            return jwt.decode(
                token,
                self._public,
                algorithms=["RS256"],
                audience=self.audience,
                issuer=self.issuer,
                options={"require": REQUIRED_CLAIMS},
            )
        except jwt.PyJWTError:
            return None
```

```python
"""Servicio auth (spec §9.1)."""

import time

import httpx
from fastapi import FastAPI
from sqlalchemy.ext.asyncio import AsyncEngine, create_async_engine

from auth_svc import admin, internal, routes
from auth_svc.config import Settings, load_settings
from auth_svc.tokens import Signer
from shared.audit_client import AuditClient
from shared.service_auth import ServiceAuth, ServiceClient
from shared.web import create_base_app


def create_app(
    settings: Settings | None = None,
    *,
    engine: AsyncEngine | None = None,
    service_auth: ServiceAuth | None = None,
    transport: httpx.AsyncBaseTransport | None = None,
) -> FastAPI:
    settings = settings or load_settings()
    service_auth = service_auth or ServiceAuth.from_secrets("auth", needs_token=True)
    peers = ServiceClient(
        service_auth, {"audit": settings.audit_url, "chat": settings.chat_url}, transport
    )
    app = create_base_app(
        title="auth",
        settings=settings,
        engine=engine or create_async_engine(settings.database_url, pool_pre_ping=True),
        service_auth=service_auth,
        audit=AuditClient(peers),
    )
    app.state.peers = peers
    app.state.signer = Signer(settings.jwt_private_key, settings.jwt_issuer, settings.jwt_audience)
    app.state.clock = time.time
    for module in (routes, internal, admin):
        app.include_router(module.router)
    return app
```

```python
    unsigned = b64e(b'{"alg":"none","typ":"JWT"}') + "." + body
    assert SIGNER.read(unsigned + ".") is None
    public_pem = (
        serialization.load_pem_private_key(JWT_KEY.encode(), password=None)
        .public_key()
        .public_bytes(serialization.Encoding.PEM, serialization.PublicFormat.SubjectPublicKeyInfo)
    )
    signing_input = b64e(b'{"alg":"HS256","typ":"JWT"}') + "." + body
    forged_sig = b64e(hmac.new(public_pem, signing_input.encode(), hashlib.sha256).digest())
    assert SIGNER.read(f"{signing_input}.{forged_sig}") is None
```

```python
"""Cuentas: normalización, política, alta y consultas (spec §8.1, §9.1)."""

import re
from uuid import uuid4

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncConnection

from shared.passwords import strong_enough

USERNAME_RE = re.compile(r"[a-z0-9_.-]{3,32}")

BY_USERNAME = text(
    "SELECT u.id::text AS id, u.username, u.password_hash, "
    "COALESCE(u.suspended_until > now(), false) AS suspended, "
    "u.totp_confirmed_at IS NOT NULL AS totp_confirmed, "
    "EXISTS (SELECT 1 FROM user_roles ur JOIN role_permissions rp ON rp.role_id = ur.role_id "
    "        JOIN permissions p ON p.code = rp.permission "
    "        WHERE ur.user_id = u.id AND p.requires_mfa) AS needs_mfa "
    "FROM users u WHERE u.username = :u"
)
ROLE_NAMES = text(
    "SELECT r.name FROM user_roles ur JOIN roles r ON r.id = ur.role_id "
    "WHERE ur.user_id = :user ORDER BY r.name"
)


def normalize(username: str) -> str:
    return username.strip().lower()


def valid_credentials(username: str, password: str) -> bool:
    """Usuario a-z0-9_.- de 3–32; contraseña ≥ 12, 3 de 4 clases y distinta del usuario (spec §8.1)."""
    return bool(USERNAME_RE.fullmatch(username)) and strong_enough(password, forbid=username)


async def insert_user(
    conn: AsyncConnection, username: str, password_hash: str, *, roles: tuple[str, ...] = ("user",)
) -> str:
    """Crea la cuenta y le asigna roles por nombre. IntegrityError si el usuario ya existe."""
    user_id = str(uuid4())
    await conn.execute(
        text("INSERT INTO users (id, username, password_hash) VALUES (:id, :u, :hash)"),
        {"id": user_id, "u": username, "hash": password_hash},
    )
    await conn.execute(
        text(
            "INSERT INTO user_roles (user_id, role_id) "
            "SELECT :id, id FROM roles WHERE name = ANY(:roles)"
        ),
        {"id": user_id, "roles": list(roles)},
    )
    return user_id


async def by_username(conn: AsyncConnection, username: str):
    return (await conn.execute(BY_USERNAME, {"u": username})).first()


async def role_names(conn: AsyncConnection, user_id: str) -> list[str]:
    return list((await conn.execute(ROLE_NAMES, {"user": user_id})).scalars())
```

```python
"""Sesiones de la cuenta: una sesión full viva por usuario, ligada a su dispositivo (spec §9.1; R13)."""

import hmac
import logging
import time
from datetime import datetime
from uuid import uuid4

import httpx
from fastapi import Response
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncConnection

from shared.web import SESSION_COOKIE, set_cookie

log = logging.getLogger("auth.sessions")

EXPIRE_STALE = text(
    "UPDATE user_sessions SET ended_at = now(), end_reason = 'expired' "
    "WHERE user_id = :user AND ended_at IS NULL "
    "AND (expires_at <= now() OR last_seen <= now() - make_interval(mins => :idle))"
)
CURRENT_FULL = text(
    "SELECT id::text AS id, device_hash FROM user_sessions "
    "WHERE user_id = :user AND stage = 'full' AND ended_at IS NULL FOR UPDATE"
)
INSERT = text(
    "INSERT INTO user_sessions (id, user_id, stage, mfa, device_hash, expires_at) "
    "VALUES (:id, :user, :stage, :mfa, :device, now() + make_interval(secs => :ttl)) "
    "RETURNING expires_at"
)
END = text(
    "UPDATE user_sessions SET ended_at = now(), end_reason = :reason "
    "WHERE id = :id AND ended_at IS NULL"
)


class SecondDevice(Exception):
    """La cuenta ya tiene una sesión full viva en otro dispositivo: se rechaza el login (spec §2)."""


async def end(conn: AsyncConnection, session_id: str, reason: str) -> None:
    await conn.execute(END, {"id": session_id, "reason": reason})


async def expire_stale(conn: AsyncConnection, settings, user_id: str) -> None:
    """Sin esto, una sesión vencida sin ended_at bloquearía el índice único (spec §7.1)."""
    await conn.execute(EXPIRE_STALE, {"user": user_id, "idle": settings.session_idle_min})


async def _insert(conn, user_id, stage, mfa, device_hash, ttl_seconds) -> tuple[str, datetime]:
    session_id = str(uuid4())
    expires_at = (
        await conn.execute(
            INSERT,
            {
                "id": session_id,
                "user": user_id,
                "stage": stage,
                "mfa": mfa,
                "device": device_hash,
                "ttl": ttl_seconds,
            },
        )
    ).scalar_one()
    return session_id, expires_at


async def create_full(
    conn: AsyncConnection, settings, user_id: str, device_hash: bytes, *, mfa: bool
) -> tuple[str, datetime]:
    """Mismo dispositivo → reemplaza la sesión anterior; otro dispositivo → SecondDevice."""
    await expire_stale(conn, settings, user_id)
    current = (await conn.execute(CURRENT_FULL, {"user": user_id})).first()
    if current is not None:
        if not hmac.compare_digest(bytes(current.device_hash), device_hash):
            raise SecondDevice
        await end(conn, current.id, "replaced")
    return await _insert(conn, user_id, "full", mfa, device_hash, settings.session_abs_hours * 3600)


async def create_preauth(
    conn: AsyncConnection, settings, user_id: str, device_hash: bytes
) -> tuple[str, datetime]:
    return await _insert(conn, user_id, "preauth", False, device_hash, settings.preauth_min * 60)


def issue(app, response: Response, *, user_id, username, session_id, stage, roles, expires_at) -> str:
    """Firma el JWT y lo deja en la cookie __Host-session (la SPA nunca lo lee desde JS)."""
    token = app.state.signer.issue(
        user_id=user_id,
        username=username,
        session_id=session_id,
        stage=stage,
        roles=roles,
        expires_at=expires_at,
    )
    set_cookie(response, SESSION_COOKIE, token, max_age=max(int(expires_at.timestamp() - time.time()), 1))
    return token


def login_body(token: str, expires_at: datetime) -> dict:
    """Para clientes de API (Postman): el mismo JWT, para mandarlo como Bearer."""
    return {
        "access_token": token,
        "token_type": "Bearer",
        "expires_in": max(int(expires_at.timestamp() - time.time()), 0),
    }


async def disconnect(app, user_id: str) -> None:
    """Pide a chat cerrar los WebSockets del usuario. Si no responde, el latido lo reconcilia (spec §4.4)."""
    try:
```

```python
"""Rutas internas de auth (spec §8.2): verify para el gateway en cada request (riesgos 5, 13 y 16)."""

import hmac

from fastapi import APIRouter, Depends, HTTPException, Request, Response
from sqlalchemy import text

from shared.crypto import token_hash
from shared.identity import client_ip
from shared.service_auth import allow
from shared.web import DEVICE_COOKIE, SESSION_COOKIE

router = APIRouter()

SESSION = text(
    "SELECT s.device_hash, s.mfa, u.username FROM user_sessions s JOIN users u ON u.id = s.user_id "
    "WHERE s.id = :sid AND s.user_id = :user AND s.stage = 'full' AND s.ended_at IS NULL "
    "AND s.expires_at > now() AND s.last_seen > now() - make_interval(mins => :idle) "
    "AND (u.suspended_until IS NULL OR u.suspended_until <= now())"
)
# Los permisos de admin (requires_mfa) solo valen en una sesión que pasó por TOTP (spec §9.1, Verify 4).
PERMS = text(
    "SELECT DISTINCT rp.permission FROM user_roles ur "
    "JOIN role_permissions rp ON rp.role_id = ur.role_id "
    "JOIN permissions p ON p.code = rp.permission "
    "WHERE ur.user_id = :user AND (CAST(:mfa AS boolean) OR NOT p.requires_mfa) ORDER BY 1"
)
TOUCH = text("UPDATE user_sessions SET last_seen = now() WHERE id = :sid")


def presented_token(request: Request) -> str | None:
    """Bearer (clientes de API, Postman) o la cookie __Host-session (navegador)."""
    header = request.headers.get("authorization", "")
    if header[:7].lower() == "bearer ":
        return header[7:].strip()
    return request.cookies.get(SESSION_COOKIE)


@router.get("/internal/verify", status_code=204, dependencies=[Depends(allow("gateway"))])
async def verify(request: Request, response: Response) -> None:
    app = request.app
    token, device = presented_token(request), request.cookies.get(DEVICE_COOKIE)
    claims = app.state.signer.read(token) if token else None
    if claims is None or claims["stage"] != "full" or not device:
        raise HTTPException(401, "no autenticado")
    async with app.state.engine.begin() as conn:
        row = (
            await conn.execute(
                SESSION,
                {"sid": claims["sid"], "user": claims["sub"], "idle": app.state.settings.session_idle_min},
            )
        ).first()
        if row is None:
            raise HTTPException(401, "no autenticado")
        mismatch = not hmac.compare_digest(bytes(row.device_hash), token_hash(device))
        if not mismatch:
            perms = list((await conn.execute(PERMS, {"user": claims["sub"], "mfa": row.mfa})).scalars())
            await conn.execute(TOUCH, {"sid": claims["sid"]})  # desliza la inactividad
    if mismatch:  # token copiado a otro equipo (riesgo 13)
        ip = client_ip(request)
        await app.state.audit.emit(
            "warning", "auth.verify.device_mismatch", actor=f"user:{row.username}",
            target=request.headers.get("x-original-uri"), ip=ip,
        )
        raise HTTPException(401, "no autenticado")
    response.headers["X-User-Id"] = claims["sub"]
    response.headers["X-User-Name"] = row.username
    response.headers["X-Session-Id"] = claims["sid"]
    if perms:
        response.headers["X-User-Perms"] = ",".join(perms)
```

```python
"""TOTP (RFC 6238) con anti-reuso: un código vale una sola vez (spec §9.1, paso 8)."""

import hmac

import pyotp

STEP_SECONDS = 30


def verify_totp(secret: str, code: str, last_step: int, now: float) -> int | None:
    """Devuelve el paso de tiempo aceptado (ventana ±1) o None. Rechaza pasos ya usados."""
    totp = pyotp.TOTP(secret)
    current = int(now) // STEP_SECONDS
    for step in (current - 1, current, current + 1):
        if step > last_step and hmac.compare_digest(totp.at(step * STEP_SECONDS), code):
            return step
    return None
```

```python
            },
        )
```

```python
async def totp(body: TotpCode, request: Request, response: Response) -> dict:
    settings, engine, audit = request.app.state.settings, request.app.state.engine, request.app.state.audit
    ip, claims = client_ip(request), preauth_claims(request)
    device = token_hash(request.cookies.get(DEVICE_COOKIE, ""))
    outcome, row = "invalid", None
    try:
        async with engine.begin() as conn:
            if claims is not None:
                row = (await conn.execute(PREAUTH, {"sid": claims["sid"], "user": claims["sub"]})).first()
            # La etapa preauth vale solo en el dispositivo donde se puso la contraseña.
            if row is not None and row.totp_secret is not None and hmac.compare_digest(bytes(row.device_hash), device):
                limits = login_limits(settings, row.username, ip)
                if await ratelimit.any_over(conn, limits, settings.login_window_min):
                    outcome = "locked"
                else:
                    secret = unseal(
                        settings.keks[row.kek_version], bytes(row.totp_secret), totp_aad(row.user_id)
                    ).decode()
                    step = verify_totp(secret, body.code, row.totp_last_step, request.app.state.clock())
                    if step is None:
                        await ratelimit.record(conn, [k for k, _ in limits], settings.login_window_min)
                    else:
                        await conn.execute(
                            text(
                                "UPDATE users SET totp_last_step = :step, "
                                "totp_confirmed_at = COALESCE(totp_confirmed_at, now()) WHERE id = :id"
                            ),
                            {"step": step, "id": row.user_id},
                        )
                        await sessions.end(conn, claims["sid"], "mfa_passed")
                        session_id, expires_at = await sessions.create_full(
                            conn, settings, row.user_id, device, mfa=True
                        )
                        roles = await accounts.role_names(conn, row.user_id)
                        if not row.confirmed:
                            await audit.emit("info", "auth.totp.enrolled", actor=f"user:{row.username}", ip=ip)
                        await audit.emit(
                            "info", "auth.login.ok", actor=f"user:{row.username}", ip=ip,
                            details={"mfa": True}, required=True,
                        )
                        outcome = "ok"
    except (sessions.SecondDevice, IntegrityError):  # se revierte: el código no queda gastado
        await audit.emit("warning", "auth.login.second_device", actor=f"user:{row.username}", ip=ip)
        raise second_device() from None
    if outcome == "ok":
        token = sessions.issue(
            request.app, response, user_id=row.user_id, username=row.username, session_id=session_id,
            stage="full", roles=roles, expires_at=expires_at,
        )
        return sessions.login_body(token, expires_at)
    target = row.username if row is not None else None
    if outcome == "locked":
        await audit.emit("warning", "auth.login.locked", actor=f"ip:{ip}", target=target, ip=ip)
        raise too_many(settings)
```

```python
    "SELECT r.id::text AS id, r.name, r.description, r.builtin, "
    "COALESCE(array_agg(rp.permission ORDER BY rp.permission) "
    "         FILTER (WHERE rp.permission IS NOT NULL), ARRAY[]::text[]) AS permissions "
    "FROM roles r LEFT JOIN role_permissions rp ON rp.role_id = r.id "

)

    "u.suspended_until::text AS suspended_until, u.created_at, "
    "COALESCE(array_agg(r.name ORDER BY r.name) FILTER (WHERE r.name IS NOT NULL), ARRAY[]::text[]) AS roles "
    "FROM users u LEFT JOIN user_roles ur ON ur.user_id = u.id LEFT JOIN roles r ON r.id = ur.role_id "
    "GROUP BY u.id ORDER BY u.username LIMIT 200"
)

    "INSERT INTO role_permissions (role_id, permission) SELECT :id, unnest(CAST(:perms AS text[]))"
)

    blocked = False
    async with request.app.state.engine.begin() as conn:
        username = (await conn.execute(text("SELECT username FROM users WHERE id = :id FOR UPDATE"), {"id": uid})).scalar()

            raise HTTPException(404, "usuario no encontrado")
```

```python
@router.post("/api/admin/users/{user_id}/suspend", status_code=204)
async def suspend(
    user_id: UUID, body: SuspendIn, request: Request, who: UserIdentity = Depends(require_perm("users:manage"))
) -> None:
    uid = str(user_id)
    if uid == who.user_id:
        raise HTTPException(409, "no puedes suspenderte a ti mismo")
    blocked = False
    async with request.app.state.engine.begin() as conn:
        username = (await conn.execute(text("SELECT username FROM users WHERE id = :id FOR UPDATE"), {"id": uid})).scalar()
        if username is None:
            raise HTTPException(404, "usuario no encontrado")
        if await has_role(conn, uid, "admin"):
            blocked = await remaining_admins(conn, excluding=uid) == 0
        if not blocked:
            await conn.execute(
                text(
                    "UPDATE users SET suspended_reason = :reason, suspended_until = CASE "
                    "WHEN CAST(:minutes AS int) IS NULL THEN 'infinity'::timestamptz "
                    "ELSE now() + make_interval(mins => CAST(:minutes AS int)) END WHERE id = :id"
                ),
                {"reason": body.reason, "minutes": body.minutes, "id": uid},
            )
            await sessions.end_all(conn, uid, "suspended")
    if blocked:
        await request.app.state.audit.emit("warning", "admin.guard.last_admin", actor=f"user:{who.username}", target=username)
        raise HTTPException(409, "no puede quedar el sistema sin admin")
    # Quitar acceso nunca espera a la auditoría (spec §14.4).
    await request.app.state.audit.emit(
        "warning", "user.suspended", actor=f"user:{who.username}", target=username,
        details={"reason": body.reason, "minutes": body.minutes},

    await sessions.disconnect(request.app, uid)
```

```python
"""Re-envuelve los secretos TOTP con la KEK vigente (spec §11.3).

Procedimiento: montar kek_auth_v{n+1}, poner CURRENT_KEK_VERSION=n+1, reiniciar auth y ejecutar
    docker compose -f deploy/docker-compose.yml exec auth python -m auth_svc.rotate_kek
Después ya se puede retirar el secreto de la versión anterior.
"""

import asyncio

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncEngine, create_async_engine

from auth_svc.config import Settings, load_settings
from shared.audit_client import AuditClient
from shared.crypto import seal, totp_aad, unseal
from shared.service_auth import ServiceAuth, ServiceClient


async def rotate(engine: AsyncEngine, settings: Settings, audit) -> int:
    target = settings.current_kek_version
    async with engine.begin() as conn:
        rows = (
            await conn.execute(
                text(
                    "SELECT id::text AS id, totp_secret, kek_version FROM users "
                    "WHERE totp_secret IS NOT NULL AND kek_version <> :v FOR UPDATE"
                ),
                {"v": target},
            )
        ).all()
        for row in rows:
            aad = totp_aad(row.id)
            secret = unseal(settings.keks[row.kek_version], bytes(row.totp_secret), aad)
            await conn.execute(
                text("UPDATE users SET totp_secret = :secret, kek_version = :v WHERE id = :id"),
                {"secret": seal(settings.kek, secret, aad), "v": target, "id": row.id},
            )
        if rows:
            await audit.emit(
                "info", "key.rotate.auth", actor="system", details={"to": target, "users": len(rows)},
                required=True,
            )
    return len(rows)


async def _main() -> int:
    settings = load_settings()
    engine = create_async_engine(settings.database_url)
    peers = ServiceClient(ServiceAuth.from_secrets("auth", needs_token=True), {"audit": settings.audit_url})
    try:
        return await rotate(engine, settings, AuditClient(peers))
    finally:
        await engine.dispose()


if __name__ == "__main__":
    count = asyncio.run(_main())
    print(f"{count} secretos TOTP re-envueltos con la KEK vigente")
```
