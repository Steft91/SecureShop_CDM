# 03 · Servicio `auth` — Plan de implementación (rev. 4)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Cuentas con registro abierto, roles dinámicos con permisos de un catálogo fijo (RBAC), login con Argon2id, bloqueo y retraso progresivo, un JWT RS256 que apunta a una sesión revocable ligada a un único dispositivo, y el endpoint `verify` que el gateway llama en cada request para obtener la identidad y los permisos **actuales** (riesgos R4, R5, R13 y R16). El admin global se crea solo por CLI: no hay credenciales por defecto.

**Architecture:**
- Rutas públicas en `routes.py` (`/api/auth/*`) y de administración en `admin.py` (`/api/admin/*`); la interna `/internal/verify` (solo para el gateway) en `internal.py`.
- `tokens.Signer` firma y verifica los JWT con el algoritmo fijo `RS256`. El JWT nunca se guarda: la BD solo tiene la sesión (`user_sessions`), que decide revocación, inactividad y dispositivo.
- `sessions.py` aplica la política de dispositivo único: un índice único parcial impide dos sesiones vivas por cuenta, y el login desde otro dispositivo se rechaza.
- Eventos de seguridad con `shared.audit.emit` (líneas JSON en la salida del contenedor): los que otorgan algo van después del commit; los de rechazo, donde se decide el rechazo.

**Tech Stack:** Python 3.13, FastAPI, SQLAlchemy async, Alembic, argon2-cffi (vía `shared.passwords`), PyJWT, cryptography, pytest.

**Spec:** `docs/specs/2026-10-12-chat-seguro-design.md` §7.1, §8.1–§8.2, §9.1, §11, §16.4 (R4, R5, R11, R13, R16).

## Global Constraints

Las del índice (`2026-10-13-00-indice.md`). En particular:
- `text()` con parámetros nombrados.
- Todas las rutas exigen `X-Service-Token` salvo `/health`.
- Errores `{"error": "..."}` genéricos; cuerpos con `extra="forbid"`.
- Cookies `__Host-session` (JWT) y `__Host-device`, con `Secure`, `HttpOnly` y `SameSite=Strict`.
- Argon2id con ≤ 4 cálculos concurrentes.
- JWT solo con `algorithms=["RS256"]`; claims `iss`, `aud`, `sub`, `sid`, `name`, `roles`, `iat`, `exp`.
- Commits en español, sin atribución.

```python
"""JWT RS256 que apunta a una sesión del servidor (spec §8.1; riesgo 16).

El algoritmo va fijo al verificar: un token con `alg: none`, o uno firmado con HS256 usando la
clave pública como secreto, no pasa. La revocación no depende del `exp`: la decide la sesión (verify).
"""

import time
from datetime import datetime

import jwt
from cryptography.hazmat.primitives import serialization

REQUIRED_CLAIMS = ["iss", "aud", "sub", "sid", "iat", "exp"]


class Signer:
    def __init__(self, private_pem: str, issuer: str, audience: str) -> None:
        self._private = serialization.load_pem_private_key(private_pem.encode(), password=None)
        self._public = self._private.public_key()
        self.issuer, self.audience = issuer, audience

    def issue(
        self,
        *,
        user_id: str,
        username: str,
        session_id: str,
        roles: list[str],
        expires_at: datetime,
    ) -> str:
        claims = {
            "iss": self.issuer,
            "aud": self.audience,
            "sub": user_id,
            "sid": session_id,
            "name": username,
            "roles": roles,  # solo informativo: la autorización usa los permisos de la BD
            "iat": int(time.time()),
            "exp": int(expires_at.timestamp()),
        }
        return jwt.encode(claims, self._private, algorithm="RS256")

    def read(self, token: str) -> dict | None:
        """Claims de un JWT con firma, emisor, audiencia y vencimiento válidos; None en otro caso."""
        try:
            return jwt.decode(
                token,
                self._public,
                algorithms=["RS256"],
                audience=self.audience,
                issuer=self.issuer,
                options={"require": REQUIRED_CLAIMS},
            )
        except jwt.PyJWTError:
            return None
```

```python
from fastapi import FastAPI
from sqlalchemy.ext.asyncio import AsyncEngine, create_async_engine

from auth_svc import admin, internal, routes
from auth_svc.config import Settings, load_settings
from auth_svc.tokens import Signer
from shared.service_auth import ServiceAuth
from shared.web import create_base_app


def create_app(
    settings: Settings | None = None,
    *,
    engine: AsyncEngine | None = None,
    service_auth: ServiceAuth | None = None,
) -> FastAPI:
    settings = settings or load_settings()
    app = create_base_app(
        title="auth",
        settings=settings,
        engine=engine or create_async_engine(settings.database_url, pool_pre_ping=True),
        service_auth=service_auth or ServiceAuth.from_secrets("auth"),
    )
    app.state.signer = Signer(settings.jwt_private_key, settings.jwt_issuer, settings.jwt_audience)
    for module in (routes, internal, admin):
        app.include_router(module.router)
    return app
```

`services/auth/tests/test_tokens.py`:
```python
import hashlib
import hmac
import json
from datetime import UTC, datetime, timedelta

import jwt
from conftest import JWT_KEY
from cryptography.hazmat.primitives import serialization

from auth_svc.tokens import Signer
from shared.crypto import b64e

SIGNER = Signer(JWT_KEY, "auth", "secure-chat")


def issue(**overrides) -> str:
    values = {
        "user_id": "u1",
        "username": "ana",
        "session_id": "s1",
        "roles": ["admin"],
        "expires_at": datetime.now(UTC) + timedelta(hours=1),
    }
    return SIGNER.issue(**{**values, **overrides})


def test_ida_y_vuelta_con_rs256():
    token = issue()
    claims = SIGNER.read(token)
    assert (claims["sub"], claims["sid"], claims["name"], claims["roles"]) == ("u1", "s1", "ana", ["admin"])
    assert jwt.get_unverified_header(token)["alg"] == "RS256"


def test_vencido_alterado_o_de_otra_audiencia_es_none():
    assert SIGNER.read(issue(expires_at=datetime.now(UTC) - timedelta(seconds=1))) is None
    head, body, sig = issue().split(".")
    assert SIGNER.read(f"{head}.{body}.{sig[::-1]}") is None
    assert Signer(JWT_KEY, "auth", "otra-app").read(issue()) is None
    assert SIGNER.read("no-es-un-jwt") is None


def test_alg_none_y_hs256_con_la_clave_publica_se_rechazan():
    payload = jwt.decode(issue(), options={"verify_signature": False})
    body = b64e(json.dumps(payload).encode())
    unsigned = b64e(b'{"alg":"none","typ":"JWT"}') + "." + body
    assert SIGNER.read(unsigned + ".") is None
    public_pem = (
        serialization.load_pem_private_key(JWT_KEY.encode(), password=None)
        .public_key()
        .public_bytes(serialization.Encoding.PEM, serialization.PublicFormat.SubjectPublicKeyInfo)
    )
    signing_input = b64e(b'{"alg":"HS256","typ":"JWT"}') + "." + body
    forged_sig = b64e(hmac.new(public_pem, signing_input.encode(), hashlib.sha256).digest())
    assert SIGNER.read(f"{signing_input}.{forged_sig}") is None
```

```python
"""Cuentas: normalización, política, alta y consultas (spec §8.1, §9.1)."""

import re
from uuid import uuid4

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncConnection

from shared.passwords import strong_enough

USERNAME_RE = re.compile(r"[a-z0-9_.-]{3,32}")
ALIAS_RE = re.compile(r"[\w .-]{3,32}")  # \w de Python incluye letras con tilde; nada de control ni <>

BY_USERNAME = text("SELECT id::text AS id, username, password_hash FROM users WHERE username = :u")
ALIAS = text("SELECT alias FROM users WHERE id = :id")
ROLE_NAMES = text(
    "SELECT r.name FROM user_roles ur JOIN roles r ON r.id = ur.role_id "
    "WHERE ur.user_id = :user ORDER BY r.name"
)


def normalize(username: str) -> str:
    return username.strip().lower()


def valid_credentials(username: str, password: str) -> bool:
    """Usuario a-z0-9_.- de 3–32; contraseña ≥ 12, 3 de 4 clases y distinta del usuario (spec §8.1)."""
    return bool(USERNAME_RE.fullmatch(username)) and strong_enough(password, forbid=username)


def normalize_alias(alias: str) -> str:
    return " ".join(alias.split())


def valid_alias(alias: str) -> bool:
    """Alias visible (RF-20): 3–32 letras, dígitos, espacios, '_', '.' o '-'. Es solo un nombre, no una credencial."""
    return bool(ALIAS_RE.fullmatch(alias))


async def insert_user(
    conn: AsyncConnection, username: str, alias: str, password_hash: str, *, roles: tuple[str, ...] = ()
) -> str:
    """Crea la cuenta con sus roles por nombre (ninguno por defecto). IntegrityError si el usuario o el alias existen."""
    user_id = str(uuid4())
    await conn.execute(
        text("INSERT INTO users (id, username, alias, password_hash) VALUES (:id, :u, :alias, :hash)"),
        {"id": user_id, "u": username, "alias": alias, "hash": password_hash},
    )
    await conn.execute(
        text(
            "INSERT INTO user_roles (user_id, role_id) "
            "SELECT :id, id FROM roles WHERE name = ANY(:roles)"
        ),
        {"id": user_id, "roles": list(roles)},
    )
    return user_id


async def by_username(conn: AsyncConnection, username: str):
    return (await conn.execute(BY_USERNAME, {"u": username})).first()


async def role_names(conn: AsyncConnection, user_id: str) -> list[str]:
    return list((await conn.execute(ROLE_NAMES, {"user": user_id})).scalars())


async def alias_of(conn: AsyncConnection, user_id: str) -> str:
    return (await conn.execute(ALIAS, {"id": user_id})).scalar_one()
```

```python
"""Sesiones de la cuenta: una sesión viva por usuario, ligada a su dispositivo (spec §9.1; R13)."""

import hmac
import time
from datetime import datetime
from uuid import uuid4

from fastapi import Response
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncConnection

from shared.web import SESSION_COOKIE, set_cookie

EXPIRE_STALE = text(
    "UPDATE user_sessions SET ended_at = now(), end_reason = 'expired' "
    "WHERE user_id = :user AND ended_at IS NULL "
    "AND (expires_at <= now() OR last_seen <= now() - make_interval(mins => :idle))"
)
CURRENT = text(
    "SELECT id::text AS id, device_hash FROM user_sessions "
    "WHERE user_id = :user AND ended_at IS NULL FOR UPDATE"
)
INSERT = text(
    "INSERT INTO user_sessions (id, user_id, device_hash, expires_at) "
    "VALUES (:id, :user, :device, now() + make_interval(secs => :ttl)) RETURNING expires_at"
)
END = text(
    "UPDATE user_sessions SET ended_at = now(), end_reason = :reason "
    "WHERE id = :id AND ended_at IS NULL"
)


class SecondDevice(Exception):
    """La cuenta ya tiene una sesión viva en otro dispositivo: se rechaza el login (spec §2)."""


async def end(conn: AsyncConnection, session_id: str, reason: str) -> None:
    await conn.execute(END, {"id": session_id, "reason": reason})


async def expire_stale(conn: AsyncConnection, settings, user_id: str) -> None:
    """Sin esto, una sesión vencida sin ended_at bloquearía el índice único (spec §7.1)."""
    await conn.execute(EXPIRE_STALE, {"user": user_id, "idle": settings.session_idle_min})


async def create(conn: AsyncConnection, settings, user_id: str, device_hash: bytes) -> tuple[str, datetime]:
    """Mismo dispositivo → reemplaza la sesión anterior; otro dispositivo → SecondDevice."""
    await expire_stale(conn, settings, user_id)
    current = (await conn.execute(CURRENT, {"user": user_id})).first()
    if current is not None:
        if not hmac.compare_digest(bytes(current.device_hash), device_hash):
            raise SecondDevice
        await end(conn, current.id, "replaced")
    session_id = str(uuid4())
    params = {"id": session_id, "user": user_id, "device": device_hash, "ttl": settings.session_abs_hours * 3600}
    expires_at = (await conn.execute(INSERT, params)).scalar_one()
    return session_id, expires_at


def issue(app, response: Response, *, user_id, username, session_id, roles, expires_at) -> str:
    """Firma el JWT y lo deja también en la cookie __Host-session."""
    token = app.state.signer.issue(
        user_id=user_id,
        username=username,
        session_id=session_id,
        roles=roles,
        expires_at=expires_at,
    )
    set_cookie(response, SESSION_COOKIE, token, max_age=max(int(expires_at.timestamp() - time.time()), 1))
    return token


def login_body(token: str, expires_at: datetime) -> dict:
    """Para clientes de API (Postman): el mismo JWT, para mandarlo como Bearer."""
    return {
        "access_token": token,
        "token_type": "Bearer",
        "expires_in": max(int(expires_at.timestamp() - time.time()), 0),
    }
```

```python
"""Rutas públicas de auth (spec §8.1, §9.1; riesgos 4, 5 y 13)."""

import asyncio

from fastapi import APIRouter, Depends, HTTPException, Request, Response
from pydantic import BaseModel, ConfigDict, Field
from sqlalchemy.exc import IntegrityError

from auth_svc import accounts, sessions
from shared import audit, ratelimit
from shared.crypto import new_token, token_hash
from shared.identity import UserIdentity, client_ip, require_gateway, require_user
from shared.passwords import hash_secret_async, verify_secret_async
from shared.web import DEVICE_COOKIE, SESSION_COOKIE, clear_cookie, set_cookie

router = APIRouter()
DEVICE_MAX_AGE = 365 * 24 * 3600  # la cookie de dispositivo dura un año
REGISTER_WINDOW_MIN = 60


class Credentials(BaseModel):
    model_config = ConfigDict(extra="forbid")
    username: str = Field(min_length=1, max_length=64)
    password: str = Field(min_length=1, max_length=256)


class Registration(Credentials):
    alias: str = Field(min_length=1, max_length=64)


@router.post("/api/auth/register", status_code=201, dependencies=[Depends(require_gateway)])
async def register(body: Registration, request: Request) -> dict:
    settings, engine = request.app.state.settings, request.app.state.engine
    ip, username = client_ip(request), accounts.normalize(body.username)
    alias = accounts.normalize_alias(body.alias)
    key = f"register:{ip}"
    async with engine.begin() as conn:
        over = await ratelimit.failures(conn, key, REGISTER_WINDOW_MIN) >= settings.register_max_per_hour
        if not over:
            await ratelimit.record(conn, [key], REGISTER_WINDOW_MIN)  # cuenta todo intento
    if over:
        audit.emit("warning", "auth.register.ratelimited", actor=f"ip:{ip}", ip=ip)
        raise HTTPException(429, "demasiados registros; intenta más tarde", headers={"Retry-After": "3600"})
    if not (accounts.valid_credentials(username, body.password) and accounts.valid_alias(alias)):
        raise HTTPException(422, "usuario, contraseña o alias no cumplen la política")
    password_hash = await hash_secret_async(body.password)
    try:
        async with engine.begin() as conn:
            user_id = await accounts.insert_user(conn, username, alias, password_hash)  # sin roles: los da un admin
    except IntegrityError:
        raise HTTPException(409, "usuario o alias no disponible") from None
    audit.emit("info", "user.registered", actor=f"ip:{ip}", target=username, ip=ip)
    return {"id": user_id, "username": username, "alias": alias}


@router.post("/api/auth/login", dependencies=[Depends(require_gateway)])
async def login(body: Credentials, request: Request, response: Response) -> dict:
    settings, engine = request.app.state.settings, request.app.state.engine
    ip, username = client_ip(request), accounts.normalize(body.username)
    limits = [(f"user:{username}", settings.login_max_fails), (f"ip:{ip}", settings.login_max_fails)]
    async with engine.connect() as conn:
        locked = await ratelimit.any_over(conn, limits, settings.login_window_min)
        user = await accounts.by_username(conn, username)
    if locked:
        audit.emit("warning", "auth.login.locked", actor=f"ip:{ip}", target=username, ip=ip)
        raise HTTPException(
            429, "demasiados intentos", headers={"Retry-After": str(settings.login_window_min * 60)}
        )
    if not await verify_secret_async(user.password_hash if user else None, body.password):
        async with engine.begin() as conn:
            fails = await ratelimit.record(conn, [k for k, _ in limits], settings.login_window_min)
        audit.emit("warning", "auth.login.fail", actor=f"ip:{ip}", target=username, ip=ip)
        await asyncio.sleep(min(settings.login_delay_base * 2 ** (fails - 1), 4))  # retraso progresivo
        raise HTTPException(401, "credenciales inválidas")
    device = request.cookies.get(DEVICE_COOKIE) or new_token(16)  # 128 bits, sin fingerprinting
    set_cookie(response, DEVICE_COOKIE, device, max_age=DEVICE_MAX_AGE)
    try:
        async with engine.begin() as conn:
            session_id, expires_at = await sessions.create(conn, settings, user.id, token_hash(device))
            roles = await accounts.role_names(conn, user.id)
    except (sessions.SecondDevice, IntegrityError):  # IntegrityError: carrera contra el índice único
        audit.emit("warning", "auth.login.second_device", actor=f"user:{username}", ip=ip)
        raise HTTPException(409, "ya hay una sesión activa en otro dispositivo") from None
```

```python
from sqlalchemy import text

SESSION = text(
    "SELECT s.device_hash, u.username FROM user_sessions s JOIN users u ON u.id = s.user_id "
    "WHERE s.id = :sid AND s.user_id = :user AND s.ended_at IS NULL "
    "AND s.expires_at > now() AND s.last_seen > now() - make_interval(mins => :idle)"
)
PERMS = text(
    "SELECT DISTINCT rp.permission FROM user_roles ur "
    "JOIN role_permissions rp ON rp.role_id = ur.role_id WHERE ur.user_id = :user ORDER BY 1"
)
TOUCH = text("UPDATE user_sessions SET last_seen = now() WHERE id = :sid")


def presented_token(request: Request) -> str | None:
    """Bearer (clientes de API, Postman) o la cookie __Host-session."""
    header = request.headers.get("authorization", "")
    if header[:7].lower() == "bearer ":
        return header[7:].strip()
    return request.cookies.get(SESSION_COOKIE)


@router.get("/internal/verify", status_code=204, dependencies=[Depends(allow("gateway"))])
async def verify(request: Request, response: Response) -> None:
    app = request.app
    token, device = presented_token(request), request.cookies.get(DEVICE_COOKIE)
    claims = app.state.signer.read(token) if token else None
    if claims is None or not device:
        raise HTTPException(401, "no autenticado")
    async with app.state.engine.begin() as conn:
        row = (
            await conn.execute(
                SESSION,
                {"sid": claims["sid"], "user": claims["sub"], "idle": app.state.settings.session_idle_min},
            )
        ).first()
        if row is None:
            raise HTTPException(401, "no autenticado")
        mismatch = not hmac.compare_digest(bytes(row.device_hash), token_hash(device))
        if not mismatch:
            perms = list((await conn.execute(PERMS, {"user": claims["sub"]})).scalars())
            await conn.execute(TOUCH, {"sid": claims["sid"]})  # desliza la inactividad
    if mismatch:  # token copiado a otro equipo (riesgo 13)
        audit.emit(
            "warning", "auth.verify.device_mismatch", actor=f"user:{row.username}",
            target=request.headers.get("x-original-uri"), ip=client_ip(request),
        )
        raise HTTPException(401, "no autenticado")
    response.headers["X-User-Id"] = claims["sub"]
    response.headers["X-User-Name"] = row.username
    response.headers["X-Session-Id"] = claims["sid"]
    if perms:
        response.headers["X-User-Perms"] = ",".join(perms)
```

```python
@router.put("/api/admin/users/{user_id}/roles")
async def set_user_roles(
    user_id: UUID, body: RolesIn, request: Request, who: UserIdentity = Depends(require_perm("roles:manage"))
) -> dict:
    uid, wanted = str(user_id), sorted({str(role) for role in body.roles})
    blocked = False
    async with request.app.state.engine.begin() as conn:
        username = (await conn.execute(text("SELECT username FROM users WHERE id = :id FOR UPDATE"), {"id": uid})).scalar()
        if username is None:
            raise HTTPException(404, "usuario no encontrado")
        names = list(
            (
                await conn.execute(
                    text("SELECT name FROM roles WHERE id = ANY(CAST(:ids AS uuid[])) ORDER BY name"),
                    {"ids": wanted},
                )
            ).scalars()
        )
        if len(names) != len(wanted):
            raise HTTPException(422, "rol desconocido")
        if "admin" not in names and await has_role(conn, uid, "admin"):
            blocked = await remaining_admins(conn, excluding=uid) == 0
        if not blocked:
            await conn.execute(text("DELETE FROM user_roles WHERE user_id = :id"), {"id": uid})
            await conn.execute(
                text("INSERT INTO user_roles (user_id, role_id) SELECT :id, unnest(CAST(:ids AS uuid[]))"),
                {"id": uid, "ids": wanted},
            )
    if blocked:  # el rechazo se registra donde se decide (rules/backend.md)
        audit.emit("warning", "admin.guard.last_admin", actor=f"user:{who.username}", target=username)
        raise HTTPException(409, "no puede quedar el sistema sin admin")
    audit.emit("info", "user.roles.changed", actor=f"user:{who.username}", target=username, details={"roles": names})
    return {"id": uid, "username": username, "roles": names}
```
