> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Microservicio sin estado que valida imágenes (tipo real, tamaño, píxeles, decodificación), detecta esteganografía (estructura, chi², RS, entropía) y emite un veredicto: `invalida`, `problema`, `revision_manual` o `limpia`. Expone `POST /api/stego/analyze` a quien tenga el permiso `stego:analyze`; solo lo llama el gateway (riesgos R5, R6, R9, R10, R16).

**Architecture:**
- `stego.py`, `png.py` y `jpeg.py` hacen el análisis con funciones de módulo (corren en un `ProcessPoolExecutor`).
- `formats.py` es el punto de extensión (`FORMATS`) y la validación previa con Pillow.
- `verdict.py` clasifica los motivos: estructurales → `problema`, estadísticos → `revision_manual`.
- `pipeline.py` corre el análisis en otro proceso con timeout; `routes.py` tiene la API (permiso, límite por usuario y eventos de seguridad con `shared.audit.emit`).
- Sin BD: `create_base_app(engine=None)` y `/health` responde sin base.

**Tech Stack:** Python 3.13, FastAPI, Pillow, numpy, scipy, asyncio, pytest.

**Spec:** `docs/specs/2026-10-12-chat-seguro-design.md` §4.2–§4.4, §8.1, §9.9, §12, §16.4 (R5, R6, R9, R10, R16).

```python
"""Imágenes sintéticas para las pruebas, la herramienta de laboratorio y la demo (spec §12.4).

La API no las usa: viven en el paquete para que los tests y tools/stego_lab.py las importen igual.
"""

import io

import numpy as np
from PIL import Image


def natural_image(seed: int = 0, size: int = 256) -> np.ndarray:
    """Imagen "fotográfica": textura suave + ruido, pasada por JPEG q=85 como una foto de celular.

    Valores del prototipo: chi² p ≈ 0.33–0.38 y RS p̂ ≤ 0.013 (limpia).
    """
    rng = np.random.default_rng(seed)
    y, x = np.mgrid[0:size, 0:size].astype(np.float64)
    base = 110 + 60 * np.sin(x / 23) * np.cos(y / 31) + 30 * np.sin((x + y) / 47)
    rgb = np.stack([base, base * 0.8 + 20, 255 - base * 0.9], axis=-1)
    rgb = rgb + rng.normal(0, 2.0, (size, size, 3))
    buffer = io.BytesIO()
    Image.fromarray(np.clip(rgb, 0, 255).astype(np.uint8)).save(buffer, format="JPEG", quality=85)
    return np.asarray(Image.open(io.BytesIO(buffer.getvalue())).convert("RGB")).copy()


def embed_lsb(pixels: np.ndarray, rate: float, mode: str = "seq", seed: int = 1) -> np.ndarray:
    """LSB replacement con bits aleatorios (simula un mensaje cifrado). mode: 'seq' | 'random'."""
    flat = pixels.reshape(-1).copy()
    count = int(len(flat) * rate)
    rng = np.random.default_rng(seed)
    bits = rng.integers(0, 2, count, dtype=np.uint8)
    index = np.arange(count) if mode == "seq" else rng.choice(len(flat), count, replace=False)
    flat[index] = (flat[index] & 0xFE) | bits
    return flat.reshape(pixels.shape)


def png_bytes(pixels: np.ndarray, **save_kwargs) -> bytes:
    buffer = io.BytesIO()
    Image.fromarray(pixels).save(buffer, format="PNG", **save_kwargs)
    return buffer.getvalue()


def jpeg_bytes(pixels: np.ndarray, quality: int = 90, **save_kwargs) -> bytes:
    buffer = io.BytesIO()
    Image.fromarray(pixels).save(buffer, format="JPEG", quality=quality, **save_kwargs)
    return buffer.getvalue()
```

```python
"""Estadísticos de estegoanálisis LSB (spec §12.2). Funciones puras sobre arrays numpy."""

import math

import numpy as np
from scipy.stats import chi2

MASK = np.array([0, 1, 1, 0], dtype=bool)


def finite(value: float) -> float | None:
    """JSON no admite NaN ni infinitos: se devuelven como null."""
    value = float(value)
    return value if math.isfinite(value) else None


def chi2_pvalue(values: np.ndarray) -> float:
    """Ataque chi² de Westfeld–Pfitzmann: p ≈ 1 si los pares (2k, 2k+1) están igualados."""
    hist = np.bincount(values.ravel(), minlength=256).astype(np.float64)
    even, odd = hist[0::2], hist[1::2]
    expected = (even + odd) / 2
    mask = expected > 4
    dof = int(mask.sum()) - 1
    if dof < 1:
        return 0.0
    statistic = float((((even - expected) ** 2)[mask] / expected[mask]).sum())
    return float(chi2.sf(statistic, dof))


def chi2_scan(pixels: np.ndarray, min_pixels: int = 1000) -> float:
    """Máximo p sobre prefijos de 10/25/50/100 % por canal (el embedding secuencial deja el prefijo con p ≈ 1)."""
    flat = pixels.reshape(-1, pixels.shape[-1]) if pixels.ndim == 3 else pixels.reshape(-1, 1)
    best = 0.0
    for fraction in (0.1, 0.25, 0.5, 1.0):
        count = int(len(flat) * fraction)
        if count < min_pixels and fraction < 1.0:
            continue
        for channel in range(flat.shape[1]):
            best = max(best, chi2_pvalue(flat[:count, channel]))
    return best


def _groups(channel: np.ndarray) -> np.ndarray:
    """Bloques 2×2 aplanados en zigzag: (0,0) (0,1) (1,1) (1,0)."""
    h, w = channel.shape[0] // 2 * 2, channel.shape[1] // 2 * 2
    blocks = channel[:h, :w].astype(np.int16).reshape(h // 2, 2, w // 2, 2).transpose(0, 2, 1, 3)
    return blocks.reshape(-1, 4)[:, [0, 1, 3, 2]]


def _variation(groups: np.ndarray) -> np.ndarray:
    return np.abs(np.diff(groups, axis=1)).sum(axis=1)


def _rs_counts(groups: np.ndarray) -> tuple[float, float, float, float]:
    base = _variation(groups)
    positive, negative = groups.copy(), groups.copy()
    positive[:, MASK] = positive[:, MASK] ^ 1  # F1: 0↔1, 2↔3, …
    negative[:, MASK] = ((negative[:, MASK] + 1) ^ 1) - 1  # F−1: −1↔0, 1↔2, …
    fp, fn = _variation(positive), _variation(negative)
    return (fp > base).mean(), (fp < base).mean(), (fn > base).mean(), (fn < base).mean()


def rs_estimate(channel: np.ndarray) -> float:
    """Análisis RS (Fridrich, Goljan, Du 2001): tasa de embedding estimada p̂ en [0, 1].

    Degenera cerca del 100 % de embedding; ahí lo cubre el chi².
    """
    groups = _groups(channel)
    if len(groups) == 0:
        return 0.0
    rm, sm, rnm, snm = _rs_counts(groups)
    r1, s1, rn1, sn1 = _rs_counts(groups ^ 1)
    d0, d1, dn0, dn1 = rm - sm, r1 - s1, rnm - snm, rn1 - sn1
    a, b, c = 2 * (d1 + d0), dn0 - dn1 - d1 - 3 * d0, d0 - dn0
    if abs(a) < 1e-12:
        if abs(b) < 1e-12:
            return 0.0
        x = -c / b
    else:
        discriminant = b * b - 4 * a * c
        if discriminant < 0:
            return 0.0
        roots = ((-b + math.sqrt(discriminant)) / (2 * a), (-b - math.sqrt(discriminant)) / (2 * a))
        x = min(roots, key=abs)
    if abs(x - 0.5) < 1e-12:
        return 0.0
    return float(min(max(x / (x - 0.5), 0.0), 1.0))


def lsb_entropy(pixels: np.ndarray, block: int = 64) -> tuple[float, float]:
    """Entropía del plano LSB por bloques: (media, fracción de bloques con H ≥ 0.999). Métrica de apoyo."""
    planes = pixels[..., None] if pixels.ndim == 2 else pixels
    h, w = planes.shape[:2]
    bh, bw = min(block, h), min(block, w)
    nh, nw = h // bh, w // bw
    bits = (planes[: nh * bh, : nw * bw] & 1).astype(np.float64)
    p = bits.reshape(nh, bh, nw, bw, -1).mean(axis=(1, 3))
    with np.errstate(divide="ignore", invalid="ignore"):
        entropy = -(np.where(p > 0, p * np.log2(p), 0) + np.where(p < 1, (1 - p) * np.log2(1 - p), 0))
    return float(entropy.mean()), float((entropy >= 0.999).mean())
```

```python
"""PNG: estructura y estegoanálisis en píxeles (spec §12.2).

`analyze` es una función de módulo: la ejecuta un ProcessPoolExecutor.
"""

import io

import numpy as np
from PIL import Image

from stego_svc import stego

KNOWN_CHUNKS = {
    b"IHDR", b"PLTE", b"IDAT", b"IEND", b"tRNS", b"cHRM", b"gAMA", b"iCCP", b"sBIT", b"sRGB",
    b"cICP", b"mDCV", b"cLLI", b"bKGD", b"hIST", b"pHYs", b"sPLT", b"eXIf", b"tIME", b"acTL",
    b"fcTL", b"fdAT", b"tEXt", b"zTXt", b"iTXt",
}
TEXT_CHUNKS = {b"tEXt", b"zTXt", b"iTXt"}


def decoded(image: Image.Image) -> Image.Image:
    """RGBA si hay transparencia; si no, RGB. Los modos de 16 bits o enteros (I;16, I, F) se reducen a
    8 bits a mano, porque Pillow no siempre los convierte directamente a RGB."""
    if image.mode in ("I", "F") or image.mode.startswith("I;16"):
        values = np.asarray(image).astype(np.float64)
        top = 65535.0 if values.max() > 255 else 255.0
        image = Image.fromarray(np.clip(values * 255.0 / top, 0, 255).astype(np.uint8))
    has_alpha = image.mode in ("RGBA", "LA", "PA") or "transparency" in image.info
    return image.convert("RGBA" if has_alpha else "RGB")


def structure(data: bytes) -> dict:
    position, text_bytes, unknown, trailing = 8, 0, [], 0
    while position + 12 <= len(data):
        length = int.from_bytes(data[position : position + 4], "big")
        ctype = data[position + 4 : position + 8]
        end = position + 12 + length
        if ctype in TEXT_CHUNKS:
            text_bytes += length
        elif ctype not in KNOWN_CHUNKS:
            unknown.append(ctype.decode("latin-1"))
        position = end
        if ctype == b"IEND":
            trailing = max(0, len(data) - end)
            break
    return {"trailing_bytes": trailing, "text_bytes": text_bytes, "unknown_chunks": unknown}


def structural_reasons(found: dict) -> list[str]:
    """Anomalías deterministas (spec §12.2, Estructura). Son los motivos que dan `problema`."""
    reasons = []
    if found["trailing_bytes"]:
        reasons.append("datos después de IEND")
    if found["text_bytes"] > 1024:
        reasons.append("chunks de texto grandes")
    if found["unknown_chunks"]:
        reasons.append("chunks desconocidos")
    return reasons


def analyze(data: bytes, cfg: dict) -> dict:
    found = structure(data)
    with Image.open(io.BytesIO(data), formats=["PNG"]) as image:
        rgb = np.asarray(decoded(image).convert("RGB"))
    chi2_p = stego.chi2_scan(rgb)
    rs_rate = max(stego.rs_estimate(rgb[..., channel]) for channel in range(3))
    entropy_mean, entropy_high = stego.lsb_entropy(rgb)
    reasons = structural_reasons(found)
    if chi2_p >= cfg["chi2_p"]:
        reasons.append("chi² del plano LSB")
    if rs_rate >= cfg["rs_rate"]:
        reasons.append("análisis RS")
    trigger = cfg.get("entropy_trigger")
    if trigger is not None and entropy_high >= trigger:
        reasons.append("entropía del plano LSB")
    metrics = {
        **found,
        "chi2_p": stego.finite(chi2_p),
        "rs_rate": stego.finite(rs_rate),
```

```python
"""JPEG: estructura (spec §12.3). El análisis LSB en píxeles no aplica a JPEG."""


def skip_entropy(data: bytes, position: int) -> int:
    """Después de SOS: salta los datos de entropía hasta el siguiente marcador real (el relleno FF00 y RSTn no cuentan)."""
    while True:
        position = data.find(b"\xff", position)
        if position < 0 or position + 1 >= len(data):
            return len(data)
        following = data[position + 1]
        if following == 0x00 or 0xD0 <= following <= 0xD7:
            position += 2
            continue
        return position


def structure(data: bytes) -> dict:
    """Recorre los segmentos por su longitud (una miniatura EXIF con su propio FFD9 no confunde)."""
    position, metadata, end, size = 2, 0, None, len(data)
    while position + 1 < size:
        if data[position] != 0xFF:
            break
        marker = data[position + 1]
        if marker == 0xFF:  # relleno
            position += 1
            continue
        if marker == 0xD9:  # EOI
            end = position + 2
            break
        if 0xD0 <= marker <= 0xD7 or marker == 0x01:  # marcadores sin longitud
            position += 2
            continue
        if position + 4 > size:
            break
        length = int.from_bytes(data[position + 2 : position + 4], "big")
        if 0xE0 <= marker <= 0xEF or marker == 0xFE:  # APPn y COM
            metadata += length
        position += 2 + length
        if marker == 0xDA:  # SOS
            position = skip_entropy(data, position)
    return {"trailing_bytes": size - end if end is not None else 0, "metadata_bytes": metadata}


def structural_reasons(found: dict) -> list[str]:
    """Anomalías deterministas (spec §12.3). Son los motivos que dan `problema`."""
    reasons = []
    if found["trailing_bytes"] > 16:
        reasons.append("datos después de EOI")
    if found["metadata_bytes"] > 65536:
        reasons.append("metadatos excesivos")
    return reasons


def analyze(data: bytes, cfg: dict) -> dict:
    found = structure(data)
    reasons = structural_reasons(found)
    return {"suspicious": bool(reasons), "reasons": reasons, "metrics": found}
```

```python
"""Formatos aceptados y su validación (spec §12.1; riesgo 9). Punto de extensión: FORMATS."""

import io
from collections.abc import Callable
from dataclasses import dataclass

from PIL import Image

from stego_svc import jpeg, png


@dataclass(frozen=True)
class Format:
    mime: str
    ext: str
    signature: bytes
    pil_format: str
    analyze: Callable[[bytes, dict], dict]  # función de módulo: corre en otro proceso


FORMATS = {
    "image/png": Format("image/png", "png", b"\x89PNG\r\n\x1a\n", "PNG", png.analyze),
    "image/jpeg": Format("image/jpeg", "jpg", b"\xff\xd8\xff", "JPEG", jpeg.analyze),
}


def detect(data: bytes) -> Format | None:
    """Por firma (magic bytes): la extensión y el Content-Type del cliente se ignoran."""
    for fmt in FORMATS.values():
        if data.startswith(fmt.signature):
            return fmt
    return None


def validate(data: bytes, fmt: Format, max_pixels: int) -> None:
    """Lanza si no es una imagen íntegra del formato que indica la firma o si supera el tope de píxeles."""
    # formats=[...] limita a Pillow a un solo decodificador: menos superficie de ataque.
    with Image.open(io.BytesIO(data), formats=[fmt.pil_format]) as image:
        if image.width * image.height > max_pixels:  # antes de decodificar: bombas de descompresión
            raise ValueError("imagen demasiado grande")
        image.verify()
    with Image.open(io.BytesIO(data), formats=[fmt.pil_format]) as image:
        image.load()
```

```python
"""Veredicto de stego (spec §12.7): invalida | problema | revision_manual | limpia."""

from shared.crypto import sha256
from stego_svc import formats

# Anomalías deterministas: el archivo trae algo que no debería. Todo otro motivo es estadístico.
STRUCTURAL = {
    "datos después de IEND",
    "chunks de texto grandes",
    "chunks desconocidos",
    "datos después de EOI",
    "metadatos excesivos",
}


class Invalid(Exception):
    """La imagen no pasa la validación previa (spec §12.1): 422 con el motivo."""

    def __init__(self, reason: str) -> None:
        super().__init__(reason)
        self.reason = reason


def classify(reasons: list[str]) -> str:
    """Si coinciden una anomalía estructural y una estadística, gana `problema` (spec §12.7)."""
    if any(reason in STRUCTURAL for reason in reasons):
        return "problema"
    return "revision_manual" if reasons else "limpia"


def check(data: bytes, settings) -> formats.Format:
    """Validación previa; lanza Invalid con un motivo en español."""
    if len(data) > settings.upload_max_bytes:
        raise Invalid("archivo demasiado grande")
    fmt = formats.detect(data)
    if fmt is None:
        raise Invalid("formato no admitido")
    try:
        formats.validate(data, fmt, settings.max_image_pixels)
    except ValueError as exc:  # tope de píxeles: bomba de descompresión
        raise Invalid("imagen demasiado grande") from exc
    except Exception as exc:  # cualquier fallo al decodificar
        raise Invalid("imagen inválida o corrupta") from exc
    return fmt


def report(fmt: formats.Format, data: bytes, analysis: dict) -> dict:
    return {
        "verdict": classify(analysis["reasons"]),
        "format": fmt.mime,
        "sha256": sha256(data).hex(),
        "checks": analysis["metrics"],
        "motivos": analysis["reasons"],
    }
```

```python
"""Análisis aislado en otro proceso con timeout (spec §9.9, §12; riesgo 10)."""

import asyncio
from concurrent.futures import ProcessPoolExecutor

from starlette.concurrency import run_in_threadpool

from stego_svc import verdict

_POOL: ProcessPoolExecutor | None = None


def pool() -> ProcessPoolExecutor:
    # ponytail: pool global de 2 procesos. Un análisis que excede el timeout sigue corriendo hasta
    # terminar (un worker no se puede matar); si eso molesta, mover el análisis a un worker externo.
    global _POOL
    if _POOL is None:
        _POOL = ProcessPoolExecutor(max_workers=2)
    return _POOL


async def analyze(data: bytes, settings) -> dict:
    """Veredicto completo (spec §12.7). Lanza verdict.Invalid si la validación previa falla."""
    fmt = await run_in_threadpool(verdict.check, data, settings)
    loop = asyncio.get_running_loop()
    try:
        analysis = await asyncio.wait_for(
            loop.run_in_executor(pool(), fmt.analyze, data, settings.stego_config),
            settings.stego_timeout_s,
        )
    except TimeoutError:
        analysis = {"suspicious": True, "reasons": ["timeout del análisis"], "metrics": {}}
    return verdict.report(fmt, data, analysis)
```

```python
"""API de esteganografía (spec §8.1, §9.9; riesgos 5, 9, 10 y 16)."""

from fastapi import APIRouter, Depends, HTTPException, Request, UploadFile
from fastapi.responses import JSONResponse

from shared import audit
from shared.identity import UserIdentity, client_ip, require_perm
from shared.ratelimit import TokenBucket
from stego_svc import pipeline, verdict

router = APIRouter()


def bucket(app, user_id: str) -> TokenBucket:
    # ponytail: un bucket por usuario en memoria (una sola réplica); con varias, moverlo a Redis.
    if user_id not in app.state.buckets:
        per_min = app.state.settings.analyze_per_min
        app.state.buckets[user_id] = TokenBucket(per_min / 60, per_min)
    return app.state.buckets[user_id]


@router.post("/api/stego/analyze")
async def analyze(
    request: Request, file: UploadFile, who: UserIdentity = Depends(require_perm("stego:analyze"))
):
    settings, ip = request.app.state.settings, client_ip(request)
    actor = f"user:{who.username}"
    if not bucket(request.app, who.user_id).allow():
        audit.emit("warning", "stego.ratelimited", actor=actor, ip=ip)
        raise HTTPException(429, "demasiados análisis; espera un minuto")
    data = await file.read(settings.upload_max_bytes + 1)
    try:
        result = await pipeline.analyze(data, settings)
    except verdict.Invalid as exc:
        audit.emit(
            "warning", "stego.analyzed", actor=actor, ip=ip, details={"verdict": "invalida", "motivos": [exc.reason]}
        )
        return JSONResponse({"error": "archivo no permitido", "motivos": [exc.reason]}, status_code=422)
    audit.emit(
        "info" if result["verdict"] == "limpia" else "warning",
        "stego.analyzed",
        actor=actor,
        ip=ip,
        details={"sha256": result["sha256"], "verdict": result["verdict"], "motivos": result["motivos"]},
    )
    return result
```
